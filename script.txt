local function T(key) return (L[Settings.Language] and L[Settings.Language][key]) or L.en[key] or key end
local translatables = {}
local function registerText(obj,key,prefix) translatables[obj]={key=key,prefix=prefix or ""}; obj.Text=(prefix or "")..T(key) end
local function retranslateAll() for obj,d in pairs(translatables) do if obj and obj.Parent then obj.Text=d.prefix..T(d.key) end end end

ParticleFrame.Size=UDim2.new(1,0,1,0); ParticleFrame.BackgroundTransparency=1
ParticleFrame.BorderSizePixel=0; ParticleFrame.ZIndex=99; ParticleFrame.ClipsDescendants=true
local particles={}

local function clearParticles()
    for _,p in ipairs(particles) do if p.obj then p.obj:Destroy() end end
    particles={}
end

local function makeParticle()
    local kind=Settings.Settings.ParticleType
    if kind=="None" then return end
    local data={kind=kind}
    if kind=="Default" then
        local c=Instance.new("Frame",ParticleFrame); c.BackgroundTransparency=1; c.ZIndex=99
        local sz=(3+math.random()*5)*Settings.Settings.ParticleSize
        local glow=Instance.new("Frame",c)
        glow.Size=UDim2.new(0,sz*3,0,sz*3); glow.Position=UDim2.new(0.5,-sz*1.5,0.5,-sz*1.5)
        glow.BackgroundColor3=Color3.fromRGB(130+math.random(0,60),40,200+math.random(0,55))
        glow.BackgroundTransparency=0.85; glow.BorderSizePixel=0; glow.ZIndex=99
        Instance.new("UICorner",glow).CornerRadius=UDim.new(1,0)
        local core=Instance.new("Frame",c)
        core.Size=UDim2.new(0,sz,0,sz); core.Position=UDim2.new(0.5,-sz/2,0.5,-sz/2)
        core.BackgroundColor3=Color3.fromRGB(180+math.random(0,75),120,255)
        core.BackgroundTransparency=0.1+math.random()*0.3; core.BorderSizePixel=0; core.ZIndex=100
        Instance.new("UICorner",core).CornerRadius=UDim.new(1,0)
        c.Size=UDim2.new(0,sz*3,0,sz*3); c.Position=UDim2.new(math.random(),0,math.random(),0)
        data.obj=c; data.core=core
        data.vx=(math.random()-0.5)*0.0006; data.vy=(math.random()-0.5)*0.0006-0.00015
        data.pulse=math.random()*math.pi*2; data.baseAlpha=0.1+math.random()*0.3
    elseif kind=="Snow" then
        local c=Instance.new("Frame",ParticleFrame); c.BackgroundTransparency=1; c.ZIndex=99
        local sz=(2+math.random()*4)*Settings.Settings.ParticleSize
        local core=Instance.new("Frame",c)
        core.Size=UDim2.new(0,sz,0,sz); core.Position=UDim2.new(0.5,-sz/2,0.5,-sz/2)
        core.BackgroundColor3=Color3.fromRGB(235+math.random(0,20),235+math.random(0,20),255)
        core.BackgroundTransparency=0.05+math.random()*0.15; core.BorderSizePixel=0; core.ZIndex=100
        Instance.new("UICorner",core).CornerRadius=UDim.new(1,0)
        c.Size=UDim2.new(0,sz*2.5,0,sz*2.5); c.Position=UDim2.new(math.random(),0,-0.05,0)
        data.obj=c; data.vx=(math.random()-0.5)*0.0004; data.vy=0.001+math.random()*0.0012
        data.sway=math.random()*math.pi*2; data.swayAmp=0.0003+math.random()*0.0004
    elseif kind=="Rain" then
        local p=Instance.new("Frame",ParticleFrame)
        local w=(1+math.random()*0.5)*Settings.Settings.ParticleSize
        local h=(8+math.random()*10)*Settings.Settings.ParticleSize
        p.Size=UDim2.new(0,w,0,h)
        p.BackgroundColor3=Color3.fromRGB(120+math.random(0,40),160+math.random(0,40),230+math.random(0,25))
        p.BackgroundTransparency=0.15+math.random()*0.25; p.BorderSizePixel=0; p.ZIndex=99
        p.Rotation=8+math.random()*6
        Instance.new("UICorner",p).CornerRadius=UDim.new(0,2)
        p.Position=UDim2.new(math.random(),0,-0.08,0)
        data.obj=p; data.vx=0.0004+math.random()*0.0002; data.vy=0.005+math.random()*0.004
    elseif kind=="BlackHole" then
        local c=Instance.new("Frame",ParticleFrame); c.BackgroundTransparency=1; c.ZIndex=99
        local sz=(2+math.random()*4)*Settings.Settings.ParticleSize
        local core=Instance.new("Frame",c)
        core.Size=UDim2.new(0,sz,0,sz); core.Position=UDim2.new(0.5,-sz/2,0.5,-sz/2)
        core.BackgroundColor3=Color3.fromRGB(200+math.random(0,55),80,255)
        core.BackgroundTransparency=0.1; core.BorderSizePixel=0; core.ZIndex=100
        Instance.new("UICorner",core).CornerRadius=UDim.new(1,0)
        c.Size=UDim2.new(0,sz*4,0,sz*4)
        local ang=math.random()*math.pi*2; local rad=0.25+math.random()*0.35
        c.Position=UDim2.new(0.5+math.cos(ang)*rad,0,0.5+math.sin(ang)*rad,0)
        data.obj=c; data.angle=ang; data.radius=rad
        data.spiralSpeed=0.5+math.random()*0.6; data.shrinkSpeed=0.06+math.random()*0.04
    end
    table.insert(particles,data)
end

local pLimits={Default=18,Snow=25,Rain=30,BlackHole=30}
RunService.RenderStepped:Connect(function(dt)
    if not Main.Visible then return end
    local kind=Settings.Settings.ParticleType; local spd=Settings.Settings.ParticleSpeed
    if kind=="None" then clearParticles(); return end
    while #particles<(pLimits[kind] or 20) do makeParticle() end
    for i=#particles,1,-1 do
        local prt=particles[i]
        if prt.kind~=kind then prt.obj:Destroy(); table.remove(particles,i)
        else
            if kind=="Default" then
                local pos=prt.obj.Position; prt.pulse=prt.pulse+dt*1.5
                local nx=pos.X.Scale+prt.vx*dt*60*spd; local ny=pos.Y.Scale+prt.vy*dt*60*spd
                if nx<-0.08 then nx=1.08 end; if nx>1.08 then nx=-0.08 end
                if ny<-0.08 then ny=1.08 end; if ny>1.08 then ny=-0.08 end
                prt.obj.Position=UDim2.new(nx,0,ny,0)
                if prt.core then prt.core.BackgroundTransparency=prt.baseAlpha+math.sin(prt.pulse)*0.15 end
            elseif kind=="Snow" then
                prt.sway=prt.sway+dt*1.8*spd; local pos=prt.obj.Position
                local nx=pos.X.Scale+(prt.vx+math.sin(prt.sway)*prt.swayAmp)*dt*60*spd
                local ny=pos.Y.Scale+prt.vy*dt*60*spd
                if ny>1.1 then ny=-0.08; nx=math.random() end
                prt.obj.Position=UDim2.new(nx,0,ny,0)
            elseif kind=="Rain" then
                local pos=prt.obj.Position
                local nx=pos.X.Scale+prt.vx*dt*60*spd; local ny=pos.Y.Scale+prt.vy*dt*60*spd
                if ny>1.15 then ny=-0.1; nx=math.random() end
                prt.obj.Position=UDim2.new(nx,0,ny,0)
            elseif kind=="BlackHole" then
                prt.angle=prt.angle+dt*prt.spiralSpeed*spd
                prt.radius=prt.radius-dt*prt.shrinkSpeed*spd
                if prt.radius<=0.015 then prt.obj:Destroy(); table.remove(particles,i)
                else prt.obj.Position=UDim2.new(0.5+math.cos(prt.angle)*prt.radius,0,0.5+math.sin(prt.angle)*prt.radius,0) end
            end
        end
    end
end)

local function WallCheck(part)
    if not Settings.Aimbot.WallCheck then return true end
    if not part or not part.Parent then return false end
    local p=RaycastParams.new(); p.FilterType=Enum.RaycastFilterType.Exclude
    local ign={Camera}; if LocalPlayer.Character then table.insert(ign,LocalPlayer.Character) end
    p.FilterDescendantsInstances=ign; p.IgnoreWater=true
    local res=workspace:Raycast(Camera.CFrame.Position,part.Position-Camera.CFrame.Position,p)
    if res and res.Instance then return res.Instance:IsDescendantOf(part.Parent) end
    return true
end

local originalC0s = {}

local function handsGetMotors(char)
    if not char then return {} end
    local motors = {}
    local armNames = {
        "RightUpperArm","RightLowerArm","RightHand",
        "LeftUpperArm","LeftLowerArm","LeftHand",
        "Right Arm","Left Arm"
    }
    for _, j in ipairs(char:GetDescendants()) do
        if j:IsA("Motor6D") then
            for _, name in ipairs(armNames) do
                if j.Part1 and j.Part1.Name == name then
                    table.insert(motors, j)
                    break
                end
            end
        end
    end
    return motors
end

local function handsSaveOriginals(char)
    originalC0s = {}
    local motors = handsGetMotors(char)
    for _, m in ipairs(motors) do
        originalC0s[m] = m.C0
    end
end

function handsResetAll()
    for m, c0 in pairs(originalC0s) do
        pcall(function() if m and m.Parent then m.C0 = c0 end end)
    end
end

local handsSaved = false
LocalPlayer.CharacterAdded:Connect(function(char)
    handsSaved = false
    originalC0s = {}
    task.wait(1)
    if Settings.Misc.HandsEnabled then
        handsSaveOriginals(char)
        handsSaved = true
    end
end)

RunService.RenderStepped:Connect(function()
    if not Settings.Misc.HandsEnabled then return end
    local char = LocalPlayer.Character
    if not char then return end

    if not handsSaved then
        handsSaveOriginals(char)
        handsSaved = true
    end

    local hx = Settings.Misc.HandsX / 30
    local hy = Settings.Misc.HandsY / 30
    local hz = Settings.Misc.HandsZ / 30
    local offset = CFrame.new(hx, hy, hz)

    for m, origC0 in pairs(originalC0s) do
        pcall(function()
            if m and m.Parent then
                local isLeft = m.Part1 and (m.Part1.Name:find("Left") ~= nil)
                if isLeft then
                    m.C0 = origC0 * CFrame.new(-hx, hy, hz)
                else
                    m.C0 = origC0 * offset
                end
            end
        end)
    end
end)

    ring.CFrame = CFrame.new(position) * CFrame.Angles(0, 0, math.rad(90))
    ring.Material = Enum.Material.Neon; ring.Color = Settings.ESP.RingStart
    ring.Transparency = 0; ring.Parent = workspace
    table.insert(jumpRings, {part = ring, life = 0, maxLife = 1.5})
end

-- ===== MOVEMENT TRAIL SYSTEM =====
local trailParts = {}
local function createTrailPart(position)
    local tp = Instance.new("Part")
    tp.Name = "FleexiTrail"; tp.Anchored = true; tp.CanCollide = false
    tp.CastShadow = false; tp.Size = Vector3.new(0.3, 0.3, 0.3)
    tp.Shape = Enum.PartType.Ball; tp.Position = position
    tp.Material = Enum.Material.Neon; tp.Color = Settings.ESP.TrailStart
    tp.Transparency = 0; tp.Parent = workspace
    table.insert(trailParts, {part = tp, life = 0, maxLife = 2})
end

local trailTimer = 0
RunService.Heartbeat:Connect(function(dt)
    if Settings.ESP.JumpRing then
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and plr.Character then
                local hum = plr.Character:FindFirstChildOfClass("Humanoid")
                local root = plr.Character:FindFirstChild("HumanoidRootPart")
                if hum and root and hum:GetState() == Enum.HumanoidStateType.Jumping then
                    if not plr:GetAttribute("FleexiJumped") then
                        plr:SetAttribute("FleexiJumped", true)
                        createJumpRing(root.Position - Vector3.new(0, 3, 0))
                    end
                else plr:SetAttribute("FleexiJumped", nil) end
            end
        end
    end

    for i = #jumpRings, 1, -1 do
        local r = jumpRings[i]; r.life = r.life + dt
        if r.life >= r.maxLife then r.part:Destroy(); table.remove(jumpRings, i)
        else
            local pct = r.life / r.maxLife; local sz = 2 + pct * 8
            r.part.Size = Vector3.new(0.1, sz, sz)
            r.part.Transparency = pct * (Settings.ESP.RingFade + 0.2)
            r.part.Color = Settings.ESP.RingStart
        end
    end

    if Settings.ESP.MovementTrail then
        trailTimer = trailTimer + dt
        if trailTimer >= 0.08 then
            trailTimer = 0
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= LocalPlayer and plr.Character then
                    local hum = plr.Character:FindFirstChildOfClass("Humanoid")
                    local root = plr.Character:FindFirstChild("HumanoidRootPart")
                    if hum and root and hum.MoveDirection.Magnitude > 0.1 then
                        createTrailPart(root.Position - Vector3.new(0, 2.5, 0))
                    end
                end
            end
        end
    end

    for i = #trailParts, 1, -1 do
        local t = trailParts[i]; t.life = t.life + dt
        if t.life >= t.maxLife then t.part:Destroy(); table.remove(trailParts, i)
        else
            local pct = t.life / t.maxLife
            t.part.Transparency = pct * Settings.ESP.TrailFade
            t.part.Color = Settings.ESP.TrailStart
            local sz = 0.3 * (1 - pct * 0.5)
            t.part.Size = Vector3.new(sz, sz, sz)
        end
    end
end)

RunService.RenderStepped:Connect(function()
    local viewport=Camera.ViewportSize
    local vx=viewport.X; local vy=viewport.Y
    local cx=vx/2; local cy=vy/2; local center2=Vector2.new(cx,cy)

    -- FOV Circle
    if Settings.Aimbot.Enabled and Settings.Aimbot.ShowFOV and not Settings.Aimbot.Mode360 then
        FOVFrame.Visible=true; local r=Settings.Aimbot.FOV
        for i=1,FOV_SEGMENTS do
            fovSegs[i].BackgroundColor3=Settings.Aimbot.FovColor
            local a1=(i-1)/FOV_SEGMENTS*math.pi*2; local a2=i/FOV_SEGMENTS*math.pi*2
            setSegLine(fovSegs[i],cx+math.cos(a1)*r,cy+math.sin(a1)*r,cx+math.cos(a2)*r,cy+math.sin(a2)*r)
        end
        if Settings.Aimbot.FovFill then
            fovFillFrame.Visible=true; fovFillFrame.Size=UDim2.new(0,r*2,0,r*2)
            fovFillFrame.Position=UDim2.new(0,cx,0,cy)
            fovFillFrame.BackgroundColor3=Settings.Aimbot.FovFillColor; fovFillFrame.BackgroundTransparency=0.85
        else fovFillFrame.Visible=false end
    else
        FOVFrame.Visible=false; fovFillFrame.Visible=false
        for i=1,FOV_SEGMENTS do fovSegs[i].Visible=false end
    end

    -- Aimbot
    if aimActive() then
        local best,bestD=nil,math.huge
        local fovLimit=Settings.Aimbot.Mode360 and 99999 or Settings.Aimbot.FOV
        for _,plr in ipairs(Players:GetPlayers()) do
            if plr~=LocalPlayer and plr.Character then
                if not(Settings.ESP.TeamCheck and plr.Team and LocalPlayer.Team and plr.Team==LocalPlayer.Team) then
                    local target=getTargetPart(plr.Character)
                    local hum=plr.Character:FindFirstChildOfClass("Humanoid")
                    if target and hum and hum.Health>0 then
                        local sp,onScreen=Camera:WorldToViewportPoint(target.Position)
                        if sp.Z>0 then
                            local d
                            if Settings.Aimbot.Mode360 then
                                local myRoot=LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                                d=myRoot and (myRoot.Position-target.Position).Magnitude or 9999
                            else
                                d=onScreen and (Vector2.new(sp.X,sp.Y)-center2).Magnitude or 99999
                            end
                            if d<=fovLimit and d<bestD and WallCheck(target) then bestD=d; best=plr end
                        end
                    end
                end
            end
        end
        if best and best.Character then
            local target=getTargetPart(best.Character)
            if target then
                Camera.CFrame=Camera.CFrame:Lerp(CFrame.new(Camera.CFrame.Position,target.Position),math.clamp(Settings.Aimbot.Smooth,0.005,1))
            end
        end
    end

    -- ESP
    local activePlrs={}
    for _,plr in ipairs(Players:GetPlayers()) do if plr~=LocalPlayer then activePlrs[plr]=true end end
    for plr,_ in pairs(ESPObjects) do if not activePlrs[plr] then DestroyESP(ESPObjects[plr]); ESPObjects[plr]=nil end end

    for plr,_ in pairs(activePlrs) do
        if not ESPObjects[plr] then CreateESP(plr) end
        local o=ESPObjects[plr]; if not o then continue end
        local char=plr.Character
        local skipTeam=Settings.ESP.TeamCheck and plr.Team and LocalPlayer.Team and plr.Team==LocalPlayer.Team

        if not(char and Settings.ESP.Enabled) or skipTeam then
            HideESP(o); UpdateChams(plr,false); UpdateAura(plr,false)
        else
            local root=char:FindFirstChild("HumanoidRootPart")
            local head=char:FindFirstChild("Head")
            local hum=char:FindFirstChildOfClass("Humanoid")

            if not(root and head and hum and hum.Health>0) then
                HideESP(o); UpdateChams(plr,false); UpdateAura(plr,false)
            else
                local myRoot=LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                local dist=myRoot and math.floor((myRoot.Position-root.Position).Magnitude) or 0

                if dist>Settings.ESP.MaxDistance then
                    HideESP(o); UpdateChams(plr,false); UpdateAura(plr,false)
                else
                    local rootCF=root.CFrame
                    local corners={
                        Vector3.new(2,3,1.5),Vector3.new(-2,3,1.5),Vector3.new(2,-3,1.5),Vector3.new(-2,-3,1.5),
                        Vector3.new(2,3,-1.5),Vector3.new(-2,3,-1.5),Vector3.new(2,-3,-1.5),Vector3.new(-2,-3,-1.5)
                    }
                    local minX,minY=math.huge,math.huge; local maxX,maxY=-math.huge,-math.huge; local anyInFront=false
                    for _,c in ipairs(corners) do
                        local wp=rootCF:PointToWorldSpace(c); local sp=Camera:WorldToViewportPoint(wp)
                        if sp.Z>0 then
                            anyInFront=true
                            if sp.X<minX then minX=sp.X end; if sp.Y<minY then minY=sp.Y end
                            if sp.X>maxX then maxX=sp.X end; if sp.Y>maxY then maxY=sp.Y end
                        end
                    end

                    -- Off-screen arrows
                    if not anyInFront and Settings.ESP.OffScreenArrows then
                        local sp3 = Camera:WorldToViewportPoint(root.Position)
                        local dir = Vector2.new(sp3.X - cx, sp3.Y - cy)
                        if dir.Magnitude > 0 then
                            dir = dir.Unit
                            local arrowDist = math.min(vx, vy) * 0.4
                            o.Arrow.Position = UDim2.new(0, cx + dir.X * arrowDist - 15, 0, cy + dir.Y * arrowDist - 15)
                            o.Arrow.Rotation = math.deg(math.atan2(dir.Y, dir.X))
                            o.Arrow.TextColor3 = Settings.ESP.ArrowColor; o.Arrow.Visible = true
                        else o.Arrow.Visible = false end
                    else o.Arrow.Visible = false end

                    if not anyInFront then
                        HideESP(o); o.Arrow.Visible = Settings.ESP.OffScreenArrows and o.Arrow.Visible
                        UpdateChams(plr,true); UpdateAura(plr,true)
                    else
                        o.Arrow.Visible = false
                        local bW=math.max(maxX-minX,8); local bH=math.max(maxY-minY,12)
                        local bX=minX; local bY=minY
                        local tSize = Settings.ESP.ESPTextScale
                        local rounding = Settings.ESP.BoxRounding

                        -- Box
                        if Settings.ESP.Boxes then
                            o.Box.Visible=true; o.Box.Size=UDim2.new(0,bW,0,bH); o.Box.Position=UDim2.new(0,bX,0,bY)
                            o.BoxStroke.Color=Settings.ESP.BoxColor; o.BoxCorner.CornerRadius=UDim.new(0,rounding)
                            o.BoxStroke.Thickness=Settings.ESP.BoxOutline and 1.5 or 0
                        else o.Box.Visible=false end

                        -- Filled box
                        if Settings.ESP.FilledBox then
                            o.BoxFill.Visible=true; o.BoxFill.Size=UDim2.new(0,bW,0,bH); o.BoxFill.Position=UDim2.new(0,bX,0,bY)
                            o.BoxFill.BackgroundColor3=Settings.ESP.FillColor; o.BoxFill.BackgroundTransparency=Settings.ESP.FillFade
                        else o.BoxFill.Visible=false end

                        -- Name
                        if Settings.ESP.Names then
                            o.Name.Visible=true; o.Name.Size=UDim2.new(0,bW+40,0,tSize+4)
                            o.Name.Position=UDim2.new(0,bX-20,0,bY-tSize-6)
                            o.Name.Text=plr.Name; o.Name.TextColor3=Settings.ESP.NameColor; o.Name.TextSize=tSize
                        else o.Name.Visible=false end

                        -- Health
                        local maxHP=math.max(hum.MaxHealth,1); local hpPct=math.clamp(hum.Health/maxHP,0,1)
                        if Settings.ESP.Health then
                            local barW=4; local barX=bX-barW-3
                            o.HealthBG.Visible=true; o.HealthBG.Size=UDim2.new(0,barW,0,bH); o.HealthBG.Position=UDim2.new(0,barX,0,bY)
                            local barH=bH*hpPct
                            o.HealthFill.Visible=true; o.HealthFill.Size=UDim2.new(0,barW,0,barH)
                            o.HealthFill.Position=UDim2.new(0,barX,0,bY+(bH-barH))
                            o.HealthFill.BackgroundColor3=Color3.fromRGB(math.floor(255*(1-hpPct)),math.floor(200*hpPct),60)
                        else o.HealthBG.Visible=false; o.HealthFill.Visible=false end

                        -- Armor
                        if Settings.ESP.ArmorBar then
                            local barW=4; local barX=bX+bW+3
                            o.ArmorBG.Visible=true; o.ArmorBG.Size=UDim2.new(0,barW,0,bH); o.ArmorBG.Position=UDim2.new(0,barX,0,bY)
                            o.ArmorFill.Visible=true; o.ArmorFill.Size=UDim2.new(0,barW,0,bH*0.5)
                            o.ArmorFill.Position=UDim2.new(0,barX,0,bY+bH*0.5); o.ArmorFill.BackgroundColor3=Settings.ESP.ArmorColor
                        else o.ArmorBG.Visible=false; o.ArmorFill.Visible=false end

                        local bottomY = bY + bH + 2
                        if Settings.ESP.Distance then
                            o.Distance.Visible=true; o.Distance.Size=UDim2.new(0,bW+40,0,tSize+2)

    if plr == LocalPlayer then return end
    CreateESP(plr)
    plr.CharacterAdded:Connect(function()
        task.wait(0.5)
        if ESPObjects[plr] then DestroyESP(ESPObjects[plr]); ESPObjects[plr]=nil end
        CreateESP(plr)
    end)
end

    if ESPObjects[plr] then DestroyESP(ESPObjects[plr]); ESPObjects[plr]=nil end
end)
