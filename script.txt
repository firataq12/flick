local crosshairParts={}; for i=1,30 do local f=Instance.new("Frame",CrosshairContainer); f.BackgroundColor3=Color3.fromRGB(0,255,0); f.BorderSizePixel=0; f.ZIndex=55; f.Visible=false; crosshairParts[i]=f end
local chDot=Instance.new("Frame",CrosshairContainer); chDot.BackgroundColor3=Color3.fromRGB(0,255,0); chDot.BorderSizePixel=0; chDot.AnchorPoint=Vector2.new(0.5,0.5); chDot.ZIndex=56; chDot.Visible=false; Instance.new("UICorner",chDot).CornerRadius=UDim.new(1,0)
RunService.RenderStepped:Connect(function()
    for i=1,30 do
        crosshairParts[i].Visible=false
        crosshairParts[i].Rotation=0
        crosshairParts[i].AnchorPoint=Vector2.new(0,0)
    end
    chDot.Visible=false
    local ch=Settings.Crosshair
    if not ch.Enabled then return end
    local vp=Camera.ViewportSize
    local cx=math.floor(vp.X/2)+ch.OffsetX
    local cy=math.floor(vp.Y/2)+ch.OffsetY
    local sz=ch.Size; local tk=ch.Thickness; local gap=ch.Gap; local col=ch.Color

    local function drawLine(idx,x,y,w,h)
        local f=crosshairParts[idx]
        f.AnchorPoint=Vector2.new(0,0)
        f.Size=UDim2.new(0,w,0,h)
        f.Position=UDim2.new(0,x,0,y)
        f.Rotation=0
        f.BackgroundColor3=col
        f.ZIndex=55
        f.Visible=true
    end
    local function drawOutline(srcIdx,dstIdx)
        local s=crosshairParts[srcIdx]
        local o=crosshairParts[dstIdx]
        o.AnchorPoint=Vector2.new(0,0)
        o.Size=UDim2.new(0,s.Size.X.Offset+2,0,s.Size.Y.Offset+2)
        o.Position=UDim2.new(0,s.Position.X.Offset-1,0,s.Position.Y.Offset-1)
        o.Rotation=0
        o.BackgroundColor3=Color3.fromRGB(0,0,0)
        o.ZIndex=54
        o.Visible=true
    end

    if ch.Style=="Cross" then
        drawLine(1, cx-math.floor(tk/2), cy-gap-sz, tk, sz)
        drawLine(2, cx-math.floor(tk/2), cy+gap, tk, sz)
        drawLine(3, cx-gap-sz, cy-math.floor(tk/2), sz, tk)
        drawLine(4, cx+gap, cy-math.floor(tk/2), sz, tk)
        if ch.Outline then
            drawOutline(1,5); drawOutline(2,6); drawOutline(3,7); drawOutline(4,8)
        end

    elseif ch.Style=="Circle" then
        for i2=1,24 do
            local a1=(i2-1)/24*math.pi*2
            local a2=i2/24*math.pi*2
            local x1=cx+math.cos(a1)*sz; local y1=cy+math.sin(a1)*sz
            local x2=cx+math.cos(a2)*sz; local y2=cy+math.sin(a2)*sz
            local dx=x2-x1; local dy=y2-y1
            local len=math.sqrt(dx*dx+dy*dy)
            local f=crosshairParts[i2]
            f.AnchorPoint=Vector2.new(0.5,0.5)
            f.Size=UDim2.new(0,math.ceil(len+1),0,tk)
            f.Position=UDim2.new(0,(x1+x2)/2,0,(y1+y2)/2)
            f.Rotation=math.deg(math.atan2(dy,dx))
            f.BackgroundColor3=col
            f.ZIndex=55
            f.Visible=true
        end

    elseif ch.Style=="Dot" then
        chDot.Size=UDim2.new(0,sz,0,sz)
        chDot.Position=UDim2.new(0,cx,0,cy)
        chDot.BackgroundColor3=col
        chDot.Visible=true
        return

    elseif ch.Style=="Arrows" then
        drawLine(1, cx-math.floor(tk/2), cy-gap-sz, tk, sz)
        drawLine(2, cx-math.floor(sz*0.35), cy-gap-sz, math.floor(sz*0.7), tk)
        drawLine(3, cx-math.floor(tk/2), cy+gap, tk, sz)
        drawLine(4, cx-math.floor(sz*0.35), cy+gap+sz-tk, math.floor(sz*0.7), tk)
        drawLine(5, cx-gap-sz, cy-math.floor(tk/2), sz, tk)
        drawLine(6, cx-gap-sz, cy-math.floor(sz*0.35), tk, math.floor(sz*0.7))
        drawLine(7, cx+gap, cy-math.floor(tk/2), sz, tk)
        drawLine(8, cx+gap+sz-tk, cy-math.floor(sz*0.35), tk, math.floor(sz*0.7))

    elseif ch.Style=="Square" then
        drawLine(1, cx-sz, cy-sz, sz*2, tk)
        drawLine(2, cx-sz, cy+sz-tk, sz*2, tk)
        drawLine(3, cx-sz, cy-sz, tk, sz*2)
        drawLine(4, cx+sz-tk, cy-sz, tk, sz*2)
        if ch.Outline then
            drawOutline(1,5); drawOutline(2,6); drawOutline(3,7); drawOutline(4,8)
        end
    end

    if ch.Dot and ch.Style~="Dot" then
        chDot.Size=UDim2.new(0,ch.DotSize,0,ch.DotSize)
        chDot.Position=UDim2.new(0,cx,0,cy)
        chDot.BackgroundColor3=col
        chDot.Visible=true
    end

    for m, c1 in pairs(originalC1s) do
        pcall(function() if m and m.Parent then m.C1 = c1 end end)

    originalC1s = {}

local function isArmMotor(motor)
    if not motor:IsA("Motor6D") then return false end
    local p1 = motor.Part1
    if not p1 then return false end
    local n = p1.Name
    return n=="RightUpperArm" or n=="RightLowerArm" or n=="RightHand"
        or n=="LeftUpperArm"  or n=="LeftLowerArm"  or n=="LeftHand"
        or n=="Right Arm"     or n=="Left Arm"
end

LocalPlayer.CharacterAdded:Connect(function()
    originalC1s = {}
end)

RunService.Heartbeat:Connect(function()
    if not Settings.Misc.HandsEnabled then return end
    local char = LocalPlayer.Character
    if not char then return end

    for _, desc in ipairs(char:GetDescendants()) do
        if desc:IsA("Motor6D") and isArmMotor(desc) then
            if not originalC1s[desc] then
                originalC1s[desc] = desc.C1
            end
        end
    end

    local hx = Settings.Misc.HandsX / 10
    local hy = Settings.Misc.HandsY / 10
    local hz = Settings.Misc.HandsZ / 10

    for motor, origC1 in pairs(originalC1s) do
        pcall(function()
            if motor and motor.Parent and motor.Part1 then
                local name = motor.Part1.Name
                local isLeft = name:find("Left") ~= nil
                local offX = isLeft and hx or -hx
                motor.C1 = origC1 * CFrame.new(offX, hy, hz)
            end
        end)
    end
end)
