local function HideESP(esp)
    if not esp then return end
    for _, v in pairs(esp) do
        if type(v) == "table" then
            for _, line in ipairs(v) do
                if line and line.Visible ~= nil then
                    pcall(function() line.Visible = false end)
                end
            end
        elseif v and v.Visible ~= nil then
            pcall(function() v.Visible = false end)
        end
    end
end

local function RemoveESP(player)
    local esp = ESPData[player]
    if esp then
        for _, v in pairs(esp) do
            if type(v) == "table" then
                for _, line in ipairs(v) do
                    pcall(function() line:Destroy() end)
                end
            else
                pcall(function() v:Destroy() end)
            end
        end
    end
    ESPData[player] = nil
end

local function CreateESP(player)
    if not player or player == LocalPlayer then return end
    RemoveESP(player)
    ESPData[player] = CreateESPObjects(player)
end

-- Helper to set line position using frames (rotation via UIGradient Rotation)
local function SetLine(frame, x1, y1, x2, y2, thickness, color)
    local dx = x2 - x1
    local dy = y2 - y1
    local length = math.sqrt(dx*dx + dy*dy)
    local angle = math.deg(math.atan2(dy, dx))
    
    frame.Size = UDim2.new(0, length, 0, thickness or 1.5)
    frame.Position = UDim2.new(0, x1, 0, y1 - (thickness or 1.5)/2)
    frame.Rotation = angle
    frame.BackgroundColor3 = color
    frame.BackgroundTransparency = 0
end

local function HideLine(line)
    if line and line.Visible ~= nil then
        line.Visible = false
    end
end

-- ========== CHAMS ==========
local ChamsObjects = {}
local GlowObjects = {}

local function ApplyChams(player)
    if not player.Character then return end
    if ChamsObjects[player] then pcall(function() ChamsObjects[player]:Destroy() end) end
    local s, h = pcall(function()
        local hl = Instance.new("Highlight")
        hl.FillColor = C3(Settings.ChamsColor)
        hl.OutlineColor = C3(Settings.ChamsOutlineColor)
        hl.FillTransparency = 0.45
        hl.OutlineTransparency = 0
        if Settings.ChamsMode == "Visible" then hl.DepthMode = Enum.HighlightDepthMode.Occluded
        elseif Settings.ChamsMode == "Invisible" then hl.DepthMode = Enum.HighlightDepthMode.Occluded; hl.FillTransparency = 1
        elseif Settings.ChamsMode == "Wall" then hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop end
        hl.Adornee = player.Character
        hl.Parent = player.Character
        return hl
    end)
    if s and h then ChamsObjects[player] = h end
end

local function RemoveChams(player)
    if ChamsObjects[player] then pcall(function() ChamsObjects[player]:Destroy() end); ChamsObjects[player] = nil end
end

local function ApplyGlow(player)
    if GlowObjects[player] then pcall(function() GlowObjects[player]:Destroy() end); GlowObjects[player] = nil end
    if not player.Character then return end
    local root = player.Character:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local s, g = pcall(function()
        local light = Instance.new("PointLight")
        light.Color = C3(Settings.GlowColor)
        light.Brightness = 3; light.Range = 15
        light.Parent = root
        return light
    end)
    if s and g then GlowObjects[player] = g end
end

local function RemoveGlow(player)
    if GlowObjects[player] then pcall(function() GlowObjects[player]:Destroy() end); GlowObjects[player] = nil end
end

local function RefreshChams()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            RemoveChams(p); RemoveGlow(p)
            if Settings.ChamsEnabled then ApplyChams(p) end
            if Settings.GlowEnabled then ApplyGlow(p) end
        end
    end
end

local function UpdateChamsColor()
    for _, h in pairs(ChamsObjects) do
        pcall(function() h.FillColor = C3(Settings.ChamsColor); h.OutlineColor = C3(Settings.ChamsOutlineColor) end)
    end
    for _, g in pairs(GlowObjects) do pcall(function() g.Color = C3(Settings.GlowColor) end) end
end

-- ========== WALL CHECK ==========
local function CanSee(part)
    if not Settings.WallCheck then return true end
    if not part then return false end
    local origin = Camera.CFrame.Position
    local dir = (part.Position - origin)
    if dir.Magnitude < 0.1 then return true end
    dir = dir.Unit * 1000
    local ray = Ray.new(origin, dir)
    local ignore = {workspace.Terrain, LocalPlayer.Character}
    local hit = workspace:FindPartOnRayWithIgnoreList(ray, ignore)
    if not hit then return true end
    if hit:IsDescendantOf(part.Parent) then return true end
    return false
end

local HitboxMap = {
    Head = "Head", Chest = "UpperTorso", Stomach = "LowerTorso",
    Pelvis = "HumanoidRootPart", Body = "HumanoidRootPart"
}

local function GetHitboxPart(character, hitboxName)
    local pn = HitboxMap[hitboxName] or "Head"
    return character:FindFirstChild(pn) or character:FindFirstChild("Torso") or character:FindFirstChild("UpperTorso") or character:FindFirstChild("HumanoidRootPart")
end

local function GetClosestPlayer()
    local closest, closestDist, closestPos = nil, Settings.FOV, nil
    local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then continue end
        if Settings.TeamCheck and player.Team and LocalPlayer.Team and player.Team == LocalPlayer.Team then continue end
        local ch = player.Character
        if not ch then continue end
        local hu = ch:FindFirstChildOfClass("Humanoid")
        local pt = GetHitboxPart(ch, Settings.AimbotHitbox)
        if hu and hu.Health > 0 and pt then
            local sp, os = Camera:WorldToViewportPoint(pt.Position)
            if os then
                local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                if d < closestDist and CanSee(pt) then
                    closestDist = d; closest = player; closestPos = pt.Position
                end
            end
        end
    end
    return closest, closestPos
end

    end)
end

-- ========== ANTI-AIM ==========
local spinAngle = 0
local jitterFlip = 1

local function GetClosestEnemyDir()
    local ch = LocalPlayer.Character
    if not ch then return nil end
    local hrp = ch:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local cp, cd = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LocalPlayer then continue end
        local c = p.Character
        if c and c:FindFirstChild("HumanoidRootPart") then
            local d = (c.HumanoidRootPart.Position - hrp.Position).Magnitude
            if d < cd then cd = d; cp = c end
        end
    end
    if cp then return (cp.HumanoidRootPart.Position - hrp.Position).Unit end
    return nil
end

local function ApplyAntiAim(dt)
    if not Settings.AntiAimEnabled then return end
    local ch = LocalPlayer.Character
    if not ch then return end
    local hrp = ch:FindFirstChild("HumanoidRootPart")
    local hu = ch:FindFirstChildOfClass("Humanoid")
    if not hrp or not hu then return end
    local yaw, pitch, roll = 0, 0, math.rad(Settings.RollAngle)
    if Settings.ManualAA ~= "None" then
        if Settings.ManualAA == "Left" then yaw = math.rad(90)
        elseif Settings.ManualAA == "Right" then yaw = math.rad(-90)
        elseif Settings.ManualAA == "Back" then yaw = math.rad(180)
        elseif Settings.ManualAA == "Front" then yaw = 0 end
    elseif Settings.Freestanding then
        local d = GetClosestEnemyDir()
        if d then yaw = math.atan2(-d.X, -d.Z) + math.rad(90) end
    else
        if Settings.YawMode == "Spin" then
            spinAngle = (spinAngle + dt * Settings.SpinSpeed * 60) % 360
            yaw = math.rad(spinAngle)
        elseif Settings.YawMode == "Backwards" then yaw = math.rad(180)
        elseif Settings.YawMode == "Jitter" then
            jitterFlip = -jitterFlip
            yaw = math.rad(Settings.YawJitterRange * jitterFlip)
        elseif Settings.YawMode == "Random" then yaw = math.rad(math.random(-180, 180))
        elseif Settings.YawMode == "Custom" then yaw = math.rad(Settings.YawOffset) end
    end
    yaw = yaw + math.rad(Settings.YawOffset)
    if Settings.InverterActive then yaw = yaw + math.rad(180) end
    if Settings.PitchMode == "Down" then pitch = math.rad(-89)
    elseif Settings.PitchMode == "Up" then pitch = math.rad(89)
    elseif Settings.PitchMode == "Zero" then pitch = 0
    elseif Settings.PitchMode == "Fake Down" then pitch = math.rad(-45) end
    pcall(function()
        hu.AutoRotate = false
        hrp.CFrame = CFrame.new(hrp.Position) * CFrame.Angles(pitch, yaw, roll)
    end)
end

-- ========== BYPASS ==========
local function SafeBypassUpdate()
    if not Settings.BypassMovement then return end
    pcall(function()
        local ch = LocalPlayer.Character
        if not ch then return end
        local hu = ch:FindFirstChildOfClass("Humanoid")
        if not hu then return end
        if Settings.WalkSpeedBoost and hu.WalkSpeed ~= Settings.WalkSpeedValue then
            hu.WalkSpeed = Settings.WalkSpeedValue
        end
        if Settings.JumpPowerBoost and hu.JumpPower ~= Settings.JumpPowerValue then
            hu.JumpPower = Settings.JumpPowerValue
        end
    end)
end

    CreateESP(player)

local function UpdateToolESP()
    if not Settings.ToolESP then
        for _, data in pairs(ToolESPData) do if data.hl then pcall(function() data.hl:Destroy() end) end end
        ToolESPData = {} return
    end
    local found = {}
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Tool") and obj.Parent and not Players:GetPlayerFromCharacter(obj.Parent) then
            found[obj] = true
            if not ToolESPData[obj] then
                local hl = Instance.new("Highlight")
                hl.FillColor = C3(Settings.ToolESPColor)
                hl.OutlineColor = C3(Settings.ToolESPColor)
                hl.FillTransparency = 0.5
                hl.Parent = obj
                ToolESPData[obj] = {hl = hl}
            end
        end
    end
    for tool, data in pairs(ToolESPData) do
        if not found[tool] then pcall(function() data.hl:Destroy() end); ToolESPData[tool] = nil end
    end
end

RunService.RenderStepped:Connect(function(dt)
    tt = tt + dt
    local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)

    ApplyAntiAim(dt)
    SafeBypassUpdate()

    fpsFrames = fpsFrames + 1
    fpsTime = fpsTime + dt
    if fpsTime >= 0.5 then
        currentFPS = math.floor(fpsFrames / fpsTime)
        fpsFrames = 0; fpsTime = 0
    end

    -- Stats HUD
    if Settings.ShowFPS or Settings.ShowMS or Settings.FakePing then
        StatsHUD.Visible = true
        local txt = ""
        if Settings.ShowFPS then txt = "FPS: " .. currentFPS end
        if Settings.ShowMS then txt = txt .. " | MS: " .. math.floor(dt*1000) end
        if Settings.FakePing then txt = txt .. " | Ping: " .. Settings.FakePingValue end
        StatsLabel.Text = txt
    else StatsHUD.Visible = false end

    -- Movement
    local ch2 = LocalPlayer.Character
    if ch2 then
        local hu = ch2:FindFirstChildOfClass("Humanoid")
        local hrp = ch2:FindFirstChild("HumanoidRootPart")
        if hu and hrp then
            if Settings.BunnyHop and hu:GetState() == Enum.HumanoidStateType.Landed then hu.Jump = true end
            if Settings.CircleStrafe and hu.MoveDirection.Magnitude > 0 then
                hrp.CFrame = CFrame.new(hrp.Position) * CFrame.Angles(0, tt * 3, 0)
            end
            if Settings.StrafeInAir and hu:GetState() == Enum.HumanoidStateType.Freefall and hu.MoveDirection.Magnitude > 0 then
                hrp.Velocity = Vector3.new(hu.MoveDirection.X * 30, hrp.Velocity.Y, hu.MoveDirection.Z * 30)
            end
            if Settings.QuickStop and hu.MoveDirection.Magnitude < 0.1 then
                hrp.Velocity = Vector3.new(0, hrp.Velocity.Y, 0)
            end
        end
    end

    -- Compass
    if Settings.Compass and ch2 then
        local dir = Camera.CFrame.LookVector
        local ang = math.deg(math.atan2(-dir.X, -dir.Z))
        local dirName = "N"
        if ang > -22.5 and ang <= 22.5 then dirName = "N"
        elseif ang > 22.5 and ang <= 67.5 then dirName = "NE"
        elseif ang > 67.5 and ang <= 112.5 then dirName = "E"
        elseif ang > 112.5 and ang <= 157.5 then dirName = "SE"
        elseif ang > 157.5 or ang <= -157.5 then dirName = "S"
        elseif ang > -157.5 and ang <= -112.5 then dirName = "SW"
        elseif ang > -112.5 and ang <= -67.5 then dirName = "W"
        elseif ang > -67.5 and ang <= -22.5 then dirName = "NW" end
        CompassLabel.Text = dirName .. " (" .. math.floor(ang) .. "°)"
        CompassLabel.Visible = true
    else CompassLabel.Visible = false end

    -- Weather
    if Settings.WeatherEnabled and #WeatherParticles > 0 then
        for _, p in ipairs(WeatherParticles) do
            local pos = p.frame.Position
            local nx = pos.X.Scale + p.vx * dt
            local ny = pos.Y.Scale + p.vy * dt
            if ny > 1.1 then ny = -0.1; nx = math.random() end
            if nx > 1.1 then nx = 0 elseif nx < -0.1 then nx = 1 end
            p.frame.Position = UDim2.new(nx, 0, ny, 0)
        end
    end

    -- FOV Circle
    FOVFrame.Size = UDim2.new(0, Settings.FOV * 2, 0, Settings.FOV * 2)
    FOVFrame.Position = UDim2.new(0.5, -Settings.FOV, 0.5, -Settings.FOV)
    FOVFrame.Visible = Settings.ShowFOV
    FOVCircle.Thickness = Settings.FOVThickness
    FOVCircle.Color = C3(Settings.FOVColor)

    -- Crosshair
    local sizeMult = Settings.CrosshairSize / 100
    local thickness = Settings.CrosshairThickness / 100
    local opacity = Settings.CrosshairOpacity / 100
    local showCh = Settings.ShowCrosshair and opacity > 0.01
    local rotSpeed = Settings.CrosshairRotationSpeed / 100
    local dotSize = math.max(2, 4 * thickness)

    -- Hide all
    for _, d in ipairs(HeartDots) do d.Visible = false end
    for _, l in ipairs(SquareLines) do l.Visible = false end
    for _, l in ipairs(HexagonLines) do l.Visible = false end
    for _, l in ipairs(StarLines) do l.Visible = false end

    if showCh then
        CrosshairContainer.Visible = true
        if Settings.CrosshairStyle == "Heart" then
            local rot = Settings.CrosshairRotation and math.rad(heartRot) or 0
            local cx, cy = Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2
            for i, dot in ipairs(HeartDots) do
                local t = (i-1) * 0.2
                local x = 14 * (math.sin(t)^3) * sizeMult
                local y = -(11*math.cos(t) - 4*math.cos(2*t) - 2*math.cos(3*t) - math.cos(4*t)) * sizeMult
                local rv = RotateV2(Vector2.new(x, y), rot)
                dot.Size = UDim2.new(0, dotSize, 0, dotSize)
                dot.Position = UDim2.new(0, cx + rv.X - dotSize/2, 0, cy + rv.Y - dotSize/2)
                dot.BackgroundColor3 = C3(Settings.CrosshairColor)
                dot.BackgroundTransparency = 1 - opacity
                dot.Visible = true
            end
            heartRot = (heartRot + dt * 85 * rotSpeed) % 360
        elseif Settings.CrosshairStyle == "Hexagon" then
            local cx, cy = Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2
            local sz = 12 * sizeMult
            for i = 1, 6 do
                local a1 = (i-1) * (math.pi*2/6) + (Settings.CrosshairRotation and tt * 0.5 * rotSpeed or 0)
                local a2 = i * (math.pi*2/6) + (Settings.CrosshairRotation and tt * 0.5 * rotSpeed or 0)
                local x1, y1 = cx + math.cos(a1)*sz, cy + math.sin(a1)*sz
                local x2, y2 = cx + math.cos(a2)*sz, cy + math.sin(a2)*sz
                SetLine(HexagonLines[i], x1, y1, x2, y2, math.max(1, 1.5 * thickness), C3(Settings.CrosshairColor))
                HexagonLines[i].BackgroundTransparency = 1 - opacity
            end
        elseif Settings.CrosshairStyle == "Star" then
            local cx, cy = Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2
            local outerR = 14 * sizeMult
            local innerR = 6 * sizeMult
            for i = 1, 10 do
                local rotOffset = Settings.CrosshairRotation and tt * 0.8 * rotSpeed or 0
                local a1 = (i-1) * (math.pi*2/10) + rotOffset
                local a2 = i * (math.pi*2/10) + rotOffset
                local r1 = (i % 2 == 0) and innerR or outerR
                local r2 = (i % 2 == 0) and outerR or innerR
                local x1, y1 = cx + math.cos(a1)*r1, cy + math.sin(a1)*r1
                local x2, y2 = cx + math.cos(a2)*r2, cy + math.sin(a2)*r2
                SetLine(StarLines[i], x1, y1, x2, y2, math.max(1, 1.5 * thickness), C3(Settings.CrosshairColor))
                StarLines[i].BackgroundTransparency = 1 - opacity
            end
        elseif Settings.CrosshairStyle == "Cross" then
            local cx, cy = Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2
            local sz = 14 * sizeMult
            SetLine(SquareLines[1], cx, cy - sz, cx, cy, math.max(1, 1.5 * thickness), C3(Settings.CrosshairColor))
            SetLine(SquareLines[2], cx, cy, cx, cy + sz, math.max(1, 1.5 * thickness), C3(Settings.CrosshairColor))
            SetLine(SquareLines[3], cx - sz, cy, cx, cy, math.max(1, 1.5 * thickness), C3(Settings.CrosshairColor))
            SetLine(SquareLines[4], cx, cy, cx + sz, cy, math.max(1, 1.5 * thickness), C3(Settings.CrosshairColor))
            for i = 1, 4 do SquareLines[i].BackgroundTransparency = 1 - opacity end
        elseif Settings.CrosshairStyle == "Square" then
            local cx, cy = Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2
            local sz = 12 * sizeMult
            SetLine(SquareLines[1], cx - sz, cy - sz, cx + sz, cy - sz, math.max(1, 1.5 * thickness), C3(Settings.CrosshairColor))
            SetLine(SquareLines[2], cx + sz, cy - sz, cx + sz, cy + sz, math.max(1, 1.5 * thickness), C3(Settings.CrosshairColor))
            SetLine(SquareLines[3], cx + sz, cy + sz, cx - sz, cy + sz, math.max(1, 1.5 * thickness), C3(Settings.CrosshairColor))
            SetLine(SquareLines[4], cx - sz, cy + sz, cx - sz, cy - sz, math.max(1, 1.5 * thickness), C3(Settings.CrosshairColor))
            for i = 1, 4 do SquareLines[i].BackgroundTransparency = 1 - opacity end
        end
    else
        CrosshairContainer.Visible = false
    end

    -- Aimbot
    local shouldAim = false
    if Settings.AimbotEnabled then
        if Settings.AimbotMode == "Always" then shouldAim = true
        elseif Settings.AimbotMode == "Hold" then shouldAim = AimbotActive
        elseif Settings.AimbotMode == "Toggle" then shouldAim = ToggleActive end
    end
    if shouldAim then
        local target, pos = GetClosestPlayer()
        if target and pos then
            local tcf = CFrame.new(Camera.CFrame.Position, pos)
            if Settings.Smoothness <= 0 then Camera.CFrame = tcf
            else
                local a = math.clamp(dt * (22 - Settings.Smoothness) * 3, 0, 1)
                Camera.CFrame = Camera.CFrame:Lerp(tcf, a)
            end
        end
    end

    -- Radar
    if Settings.RadarEnabled and RadarFrame.Visible and ch2 then
        local lp = ch2:FindFirstChild("HumanoidRootPart")
        if lp then
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                    local dot = GetRadarDot(p)
                    local rel = p.Character.HumanoidRootPart.Position - lp.Position
                    local camCF = Camera.CFrame
                    local look = Vector3.new(camCF.LookVector.X, 0, camCF.LookVector.Z).Unit
                    local ang = math.atan2(look.X, look.Z)
                    local cs, sn = math.cos(-ang), math.sin(-ang)
                    local rx = rel.X * cs - rel.Z * sn
                    local rz = rel.X * sn + rel.Z * cs
                    local zoom = Settings.RadarZoom
                    local px = math.clamp(0.5 + (rx / zoom / 2), 0, 1)
                    local py = math.clamp(0.5 + (rz / zoom / 2), 0, 1)
                    dot.Position = UDim2.new(px, -3, py, -3)
                    dot.BackgroundColor3 = C3(Settings.ESPColor)
                    dot.Visible = true
                end
            end
        end
    else
        for _, dot in pairs(RadarDots) do dot.Visible = false end
    end

    -- Tool ESP
    toolESPTimer = toolESPTimer + dt
    if toolESPTimer > 0.5 then toolESPTimer = 0; UpdateToolESP() end

    -- ========== ESP (ScreenGui based - works everywhere) ==========
    if Settings.ESPEnabled then
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer then
                local esp = ESPData[player]
                if not esp then esp = CreateESPObjects(player); ESPData[player] = esp end
                local ch3 = player.Character
                if not ch3 then HideESP(esp); continue end
                local hu = ch3:FindFirstChildOfClass("Humanoid")
                local root = ch3:FindFirstChild("HumanoidRootPart")
                local head = ch3:FindFirstChild("Head")
                if not (hu and hu.Health > 0 and root and head) then HideESP(esp); continue end
                if Settings.WallCheck and not CanSee(root) and not Settings.ESPDormant then HideESP(esp); continue end

                local headPos = head.Position + Vector3.new(0, 0.5, 0)
                local footPos = root.Position - Vector3.new(0, root.Size.Y/2, 0)
                local headScreen, headOn = Camera:WorldToViewportPoint(headPos)
                local footScreen, footOn = Camera:WorldToViewportPoint(footPos)
                if not (headOn and footOn) then HideESP(esp); continue end

                local espSizeMult = Settings.ESPSize / 100
                local espOpacity = Settings.ESPOpacity / 100
                local boxH = math.clamp(math.abs(footScreen.Y - headScreen.Y), 10, 500)
                local boxW = math.clamp(boxH * 0.6, 8, 300)
                local cx = headScreen.X
                local top = math.min(headScreen.Y, footScreen.Y)
                local bot = math.max(headScreen.Y, footScreen.Y)
                local color = C3(Settings.ESPColor)
                local lineThickness = math.max(1, math.floor(1.5 * espSizeMult))
                local transp = 1 - espOpacity

                -- BOXES
                if Settings.ESPBoxes then
                    if Settings.ESPBoxStyle == "2D" then
                        SetLine(esp.Box[1], cx - boxW/2, top, cx + boxW/2, top, lineThickness, color)
                        SetLine(esp.Box[2], cx + boxW/2, top, cx + boxW/2, bot, lineThickness, color)
                        SetLine(esp.Box[3], cx + boxW/2, bot, cx - boxW/2, bot, lineThickness, color)
                        SetLine(esp.Box[4], cx - boxW/2, bot, cx - boxW/2, top, lineThickness, color)
                        for i = 1, 4 do esp.Box[i].BackgroundTransparency = transp; esp.Box[i].Visible = true end
                        for i = 1, 8 do HideLine(esp.Corner[i]) end
                        for i = 1, 12 do HideLine(esp.Box3D[i]) end
                    elseif Settings.ESPBoxStyle == "Corner" then
                        local cl = math.min(boxW, boxH) * 0.25
                        local corners = {
                            {cx-boxW/2, top, cx-boxW/2+cl, top}, {cx-boxW/2, top, cx-boxW/2, top+cl},
                            {cx+boxW/2, top, cx+boxW/2-cl, top}, {cx+boxW/2, top, cx+boxW/2, top+cl},
                            {cx-boxW/2, bot, cx-boxW/2+cl, bot}, {cx-boxW/2, bot, cx-boxW/2, bot-cl},
                            {cx+boxW/2, bot, cx+boxW/2-cl, bot}, {cx+boxW/2, bot, cx+boxW/2, bot-cl},
                        }
                        for i = 1, 8 do
                            SetLine(esp.Corner[i], corners[i][1], corners[i][2], corners[i][3], corners[i][4], lineThickness, color)
                            esp.Corner[i].BackgroundTransparency = transp
                            esp.Corner[i].Visible = true
                        end
                        for i = 1, 4 do HideLine(esp.Box[i]) end
                        for i = 1, 12 do HideLine(esp.Box3D[i]) end
                    elseif Settings.ESPBoxStyle == "3D" then
                        local ext = Vector3.new(2, 3, 1.5)
                        local corners3d = {
                            root.Position + Vector3.new(-ext.X, ext.Y, -ext.Z),
                            root.Position + Vector3.new(ext.X, ext.Y, -ext.Z),
                            root.Position + Vector3.new(ext.X, ext.Y, ext.Z),
                            root.Position + Vector3.new(-ext.X, ext.Y, ext.Z),
                            root.Position + Vector3.new(-ext.X, -ext.Y, -ext.Z),
                            root.Position + Vector3.new(ext.X, -ext.Y, -ext.Z),
                            root.Position + Vector3.new(ext.X, -ext.Y, ext.Z),
                            root.Position + Vector3.new(-ext.X, -ext.Y, ext.Z),
                        }
                        local screenP = {}
                        local allOn = true
                        for i, cor in ipairs(corners3d) do
                            local sp, on = Camera:WorldToViewportPoint(cor)
                            screenP[i] = Vector2.new(sp.X, sp.Y)
                            if not on then allOn = false end
                        end
                        if allOn then
                            local edges = {{1,2},{2,3},{3,4},{4,1},{5,6},{6,7},{7,8},{8,5},{1,5},{2,6},{3,7},{4,8}}
                            for i, e in ipairs(edges) do
                                SetLine(esp.Box3D[i], screenP[e[1]].X, screenP[e[1]].Y, screenP[e[2]].X, screenP[e[2]].Y, lineThickness, color)
                                esp.Box3D[i].BackgroundTransparency = transp
                                esp.Box3D[i].Visible = true
                            end
                        end
                        for i = 1, 4 do HideLine(esp.Box[i]) end
                        for i = 1, 8 do HideLine(esp.Corner[i]) end
                    end
                else
                    for i = 1, 4 do HideLine(esp.Box[i]) end
                    for i = 1, 8 do HideLine(esp.Corner[i]) end
                    for i = 1, 12 do HideLine(esp.Box3D[i]) end
                end

                -- NAMES
                if Settings.ESPNames and esp.Name then
                    esp.Name.Text = player.Name
                    esp.Name.Size = UDim2.new(0, 100, 0, 16)
                    esp.Name.Position = UDim2.new(0, cx - 50, 0, top - 18)
                    esp.Name.TextColor3 = color
                    esp.Name.TextSize = 13 * espSizeMult
                    esp.Name.TextTransparency = transp
                    esp.Name.Visible = true
                elseif esp.Name then esp.Name.Visible = false end

                -- DISTANCE
                if Settings.ESPDistance and esp.Dist and ch2 then
                    local lp = ch2:FindFirstChild("HumanoidRootPart")
                    if lp then
                        local d = math.floor((lp.Position - root.Position).Magnitude)
                        esp.Dist.Text = "[" .. d .. "m]"
                        esp.Dist.Size = UDim2.new(0, 60, 0, 14)
                        esp.Dist.Position = UDim2.new(0, cx - 30, 0, bot + 4)
                        esp.Dist.TextColor3 = color
                        esp.Dist.TextSize = 11 * espSizeMult
                        esp.Dist.TextTransparency = transp
                        esp.Dist.Visible = true
                    end
                elseif esp.Dist then esp.Dist.Visible = false end

                -- WEAPON
                if Settings.ESPWeapon and esp.Weapon then
                    local tool = ch3:FindFirstChildOfClass("Tool")
                    if tool then
                        esp.Weapon.Text = "[" .. tool.Name .. "]"
                        esp.Weapon.Size = UDim2.new(0, 80, 0, 14)
                        esp.Weapon.Position = UDim2.new(0, cx - 40, 0, bot + 20)
                        esp.Weapon.TextSize = 11 * espSizeMult
                        esp.Weapon.TextTransparency = transp
                        esp.Weapon.Visible = true
                    else esp.Weapon.Visible = false end
                elseif esp.Weapon then esp.Weapon.Visible = false end

                -- HEALTH BAR
                if Settings.ESPHealth and esp.HpBG and esp.HpFill then
                    local hp = math.clamp(hu.Health / hu.MaxHealth, 0, 1)
                    local bx = cx - boxW/2 - 5
                    esp.HpBG.Size = UDim2.new(0, 2, 0, bot - top)
                    esp.HpBG.Position = UDim2.new(0, bx, 0, top)
                    esp.HpBG.Visible = true
                    
                    esp.HpFill.Size = UDim2.new(0, 2, 0, (bot - top) * hp)
                    esp.HpFill.Position = UDim2.new(0, bx, 0, bot - (bot - top) * hp)
                    esp.HpFill.BackgroundColor3 = Color3.fromRGB(math.floor((1-hp)*220), math.floor(hp*220), 30)
                    esp.HpFill.Visible = true
                else
                    if esp.HpBG then esp.HpBG.Visible = false end
                    if esp.HpFill then esp.HpFill.Visible = false end
                end

                -- TRACERS
                if Settings.ESPTracers and esp.Tracer then
                    local startX, startY = Camera.ViewportSize.X/2, Camera.ViewportSize.Y
                    SetLine(esp.Tracer, startX, startY, cx, bot, lineThickness, color)
                    esp.Tracer.BackgroundTransparency = transp
                    esp.Tracer.Visible = true
                elseif esp.Tracer then esp.Tracer.Visible = false end

                -- SNAP LINES
                if Settings.ESPSnapLines and esp.SnapLine then
                    SetLine(esp.SnapLine, center.X, center.Y, cx, (top + bot) / 2, 1, color)
                    esp.SnapLine.BackgroundTransparency = 0.5
                    esp.SnapLine.Visible = true
                elseif esp.SnapLine then esp.SnapLine.Visible = false end

                -- SKELETON
                if Settings.ESPSkeleton then
                    local boneMap = {}
                    boneMap["Head"] = FindBone(ch3, SkelNames[1].Head)
                    boneMap["UpperTorso"] = FindBone(ch3, SkelNames[1].UpperTorso)
                    boneMap["LowerTorso"] = FindBone(ch3, SkelNames[1].LowerTorso)
                    boneMap["LeftUpperArm"] = FindBone(ch3, SkelNames[1].LeftUpperArm)
                    boneMap["RightUpperArm"] = FindBone(ch3, SkelNames[1].RightUpperArm)
                    boneMap["LeftUpperLeg"] = FindBone(ch3, SkelNames[1].LeftUpperLeg)
                    boneMap["RightUpperLeg"] = FindBone(ch3, SkelNames[1].RightUpperLeg)
                    for i, conn in ipairs(SkeletonBonesConfig) do
                        local p1 = boneMap[conn[1]]
                        local p2 = boneMap[conn[2]]
                        if p1 and p2 then
                            local s1, o1 = Camera:WorldToViewportPoint(p1.Position)
                            local s2, o2 = Camera:WorldToViewportPoint(p2.Position)
                            if o1 and o2 then
                                SetLine(esp.Skel[i], s1.X, s1.Y, s2.X, s2.Y, lineThickness, C3(Settings.SkeletonColor))
                                esp.Skel[i].BackgroundTransparency = transp
                                esp.Skel[i].Visible = true
                            else HideLine(esp.Skel[i]) end
                        else HideLine(esp.Skel[i]) end
                    end
                else
                    for i = 1, 6 do HideLine(esp.Skel[i]) end
                end
            end
        end
    else
        for _, esp in pairs(ESPData) do HideESP(esp) end
    end
end)

-- Wall Jump
local wallJumpDebounce = false
UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.KeyCode == Enum.KeyCode.Space and Settings.WallJump and not wallJumpDebounce then
        local ch = LocalPlayer.Character
        if not ch then return end
        local hrp = ch:FindFirstChild("HumanoidRootPart")
        local hu = ch:FindFirstChildOfClass("Humanoid")
        if not (hrp and hu) then return end
        if hu:GetState() == Enum.HumanoidStateType.Freefall then
            local ray = Ray.new(hrp.Position, hrp.CFrame.LookVector * 3)
            local hit = workspace:FindPartOnRayWithIgnoreList(ray, {ch})
            if hit then
                wallJumpDebounce = true
                hrp.Velocity = Vector3.new(hrp.Velocity.X, 60, hrp.Velocity.Z) - (hrp.CFrame.LookVector * 30)
                task.wait(0.3)
                wallJumpDebounce = false
            end
        end
    end
end)

UserInputService.JumpRequest:Connect(function()
    if Settings.InfinityJump then
        local ch = LocalPlayer.Character
        if ch then local hu = ch:FindFirstChildOfClass("Humanoid"); if hu then hu:ChangeState(Enum.HumanoidStateType.Jumping) end end
    end
end)
