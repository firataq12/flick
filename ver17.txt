local HitboxMap = {Head="Head",Chest="UpperTorso",Stomach="LowerTorso",Pelvis="HumanoidRootPart",Body="HumanoidRootPart"}

local function GetHitboxPart(ch, name)
    local pn = HitboxMap[name] or "Head"
    return ch:FindFirstChild(pn) or ch:FindFirstChild("Head") or ch:FindFirstChild("UpperTorso") or ch:FindFirstChild("Torso") or ch:FindFirstChild("HumanoidRootPart")
end

local function CanSee(origin, part)
    if not S.WallCheck then return true end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {LocalPlayer.Character, Camera}
    local result = workspace:Raycast(origin, (part.Position - origin), params)
    if result then
        local model = part:FindFirstAncestorWhichIsA("Model")
        return model and result.Instance:IsDescendantOf(model)
    end
    return true
end

local function GetMultipoints(part)
    if not S.Multipoint or not part then return {part.Position} end
    local sz = part.Size; local sc = S.MultipointScale/100; local cf = part.CFrame
    return {cf.Position, (cf*CFrame.new((sz.X/2)*sc,0,0)).Position, (cf*CFrame.new(-(sz.X/2)*sc,0,0)).Position,
        (cf*CFrame.new(0,(sz.Y/2)*sc,0)).Position, (cf*CFrame.new(0,-(sz.Y/2)*sc,0)).Position}
end

local function UpdateVelocityTracking()
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LocalPlayer then continue end
        local ch = p.Character; if not ch then continue end
        local hrp = ch:FindFirstChild("HumanoidRootPart"); if not hrp then continue end
        local last = PlayerLastPositions[p]
        PlayerVelocities[p] = last and (hrp.Position - last)/0.03 or Vector3.zero
        PlayerLastPositions[p] = hrp.Position
    end
end

local function PredictPosition(p, pos)
    if not S.Prediction then return pos end
    return pos + (PlayerVelocities[p] or Vector3.zero) * (S.PredictionAmount/100)
end

local function GetTargetScore(p, sd, wd)
    if S.TargetPriority == "Distance" then return wd
    elseif S.TargetPriority == "Crosshair" then return sd
    elseif S.TargetPriority == "Health" then
        local ch = p.Character; local hu = ch and ch:FindFirstChildOfClass("Humanoid")
        return hu and hu.Health or 999
    elseif S.TargetPriority == "Threat" then
        return sd*0.4 + wd*0.3 + (1000-(PlayerVelocities[p] or Vector3.zero).Magnitude*10)*0.3
    end
    return sd
end

local function IsTeammate(p)
    if not S.TeamCheck then return false end
    return p.Team and LocalPlayer.Team and p.Team == LocalPlayer.Team
end

local function GetBestTarget()
    local best, bestScore, bestPos = nil, math.huge, nil
    local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    local camPos = Camera.CFrame.Position
    local useHitbox = S.AimbotHitbox
    if S.RageEnabled and S.RageForceHitbox then useHitbox = S.RageForceHitbox end
    if S.BaimEnabled then
        local ch = LocalPlayer.Character; local hu = ch and ch:FindFirstChildOfClass("Humanoid")
        local inAir = hu and hu:GetState() == Enum.HumanoidStateType.Freefall
        if S.BaimInAir and inAir then useHitbox = "Chest" end
        if S.BaimAfterMisses > 0 and MissCount >= S.BaimAfterMisses then useHitbox = "Chest" end
    end
    if S.RageForceBaim then useHitbox = "Chest" end
    if S.StickyAim and StickyTarget and StickyTarget.Parent then
        local ch = StickyTarget.Character
        if ch then local hu = ch:FindFirstChildOfClass("Humanoid"); local pt = GetHitboxPart(ch, useHitbox)
            if hu and hu.Health > 0 and pt then local wd = (pt.Position - camPos).Magnitude
                if wd <= S.MaxDistance then
                    for _, pos in ipairs(GetMultipoints(pt)) do
                        local pp = PredictPosition(StickyTarget, pos)
                        local sp, os = Camera:WorldToViewportPoint(pp)
                        if os then local sd = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                            if sd < S.FOV * (1 + S.StickyFactor/100) and CanSee(camPos, pt) then return StickyTarget, pp end end end end end end
    end
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LocalPlayer then continue end
        if IsTeammate(p) then continue end
        local ch = p.Character; if not ch then continue end
        local hu = ch:FindFirstChildOfClass("Humanoid"); local pt = GetHitboxPart(ch, useHitbox)
        if not (hu and hu.Health > 0 and pt) then continue end
        local wd = (pt.Position - camPos).Magnitude
        if wd > S.MaxDistance then continue end
        if S.WallCheck and not CanSee(camPos, pt) then continue end
        if S.RageMultiHitbox then
            for _, hb in ipairs({"Head","Chest","Stomach","Body"}) do
                local hbp = GetHitboxPart(ch, hb); if not hbp then continue end
                for _, pos in ipairs(GetMultipoints(hbp)) do
                    local pp = PredictPosition(p, pos)
                    local sp, os = Camera:WorldToViewportPoint(pp)
                    if os then local sd = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                        if sd < S.FOV then local score = GetTargetScore(p, sd, wd)
                            if score < bestScore then bestScore = score; best = p; bestPos = pp end end end end end
        else
            for _, pos in ipairs(GetMultipoints(pt)) do
                local pp = PredictPosition(p, pos)
                local sp, os = Camera:WorldToViewportPoint(pp)
                if os then local sd = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                    if sd < S.FOV then local score = GetTargetScore(p, sd, wd)
                        if score < bestScore then bestScore = score; best = p; bestPos = pp end end end end
        end
    end
    StickyTarget = best; return best, bestPos
end

-- ═══════════ TRIGGERBOT ═══════════
local function GetTriggerbotTarget()
    local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    local camPos = Camera.CFrame.Position
    local hitbox = HitboxMap[S.TriggerbotHitbox] or "Head"
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LocalPlayer then continue end
        if S.TriggerbotTeamCheck and IsTeammate(p) then continue end
        local ch = p.Character; if not ch then continue end
        local hu = ch:FindFirstChildOfClass("Humanoid")
        local pt = ch:FindFirstChild(hitbox) or ch:FindFirstChild("Head") or ch:FindFirstChild("HumanoidRootPart")
        if not (hu and hu.Health > 0 and pt) then continue end
        local wd = (pt.Position - camPos).Magnitude
        if wd > S.TriggerbotMaxDist then continue end
        if S.TriggerbotWallCheck and not CanSee(camPos, pt) then continue end
        local sp, os = Camera:WorldToViewportPoint(pt.Position)
        if os then
            local sd = (Vector2.new(sp.X, sp.Y) - center).Magnitude
            if sd < S.TriggerbotFOV then return p, pt end
        end
    end
    return nil, nil
end

-- ═══════════ ESP DRAWING HELPERS ═══════════
local function ML(t, c)
    local ok, l = pcall(Drawing.new, "Line")
    if ok then l.Thickness = t or 1.5; l.Color = c or Color3.new(1,1,1); l.Transparency = 1; l.Visible = false; return l end
end
local function MT(s, c, ce)
    local ok, t = pcall(Drawing.new, "Text")
    if ok then t.Size = s or 13; t.Color = c or Color3.new(1,1,1); t.Font = Drawing.Fonts.UI; t.Outline = true; t.OutlineColor = Color3.new(0,0,0); t.Center = ce ~= false; t.Visible = false; return t end
end
local function MCirc(r, c, f)
    local ok, ci = pcall(Drawing.new, "Circle")
    if ok then ci.Radius = r or 3; ci.Color = c or Color3.new(1,1,1); ci.Filled = f or false; ci.Thickness = 1.5; ci.Transparency = 1; ci.Visible = false; return ci end
end
local function MQ()
    local ok, q = pcall(Drawing.new, "Quad")
    if ok then q.Visible = false; q.Filled = true; q.Transparency = 0.15; return q end
end

local function DestroyESP(p)
    local e = ESPCache[p]; if not e then return end
    local function rl(l) for _, d in ipairs(l) do if d then pcall(function() d:Remove() end) end end end
    rl(e.BoxLines); rl(e.BoxOutlines); rl(e.BoxOutlines2); rl(e.CornerLines); rl(e.CornerOutlines)
    rl(e.Box3D); rl(e.Box3DOutlines); rl(e.Skeleton); rl(e.SkeletonOut); rl(e.OffscreenArrow)
    for _, k in ipairs({"FillQuad","HealthBG","HealthBGOutline","HealthFill","HealthText","NameText","DistText","WeaponText","FlagsText","Tracer","TracerOutline","SnapLine","HeadDot","HeadDotOutline","HeadDotOutline2"}) do
        if e[k] then pcall(function() e[k]:Remove() end) end end
    ESPCache[p] = nil
end

local function HideAllESP(e)
    if not e then return end
    local function hl(l) for _, d in ipairs(l) do if d then d.Visible = false end end end
    hl(e.BoxLines); hl(e.BoxOutlines); hl(e.BoxOutlines2); hl(e.CornerLines); hl(e.CornerOutlines)
    hl(e.Box3D); hl(e.Box3DOutlines); hl(e.Skeleton); hl(e.SkeletonOut); hl(e.OffscreenArrow)
    for _, k in ipairs({"FillQuad","HealthBG","HealthBGOutline","HealthFill","HealthText","NameText","DistText","WeaponText","FlagsText","Tracer","TracerOutline","SnapLine","HeadDot","HeadDotOutline","HeadDotOutline2"}) do
        if e[k] then e[k].Visible = false end end
end

local function NewCHDot() local ok, d = pcall(Drawing.new, "Circle"); if ok then d.Filled=true; d.Visible=false; d.Transparency=1; table.insert(CrosshairDrawings, d); return d end end
local function NewCHLine() local ok, l = pcall(Drawing.new, "Line"); if ok then l.Visible=false; l.Transparency=1; table.insert(CrosshairDrawings, l); return l end end

local function GenHeart() local p={}; for t=0,2*math.pi,0.15 do table.insert(p,Vector2.new(14*(math.sin(t)^3),-(11*math.cos(t)-4*math.cos(2*t)-2*math.cos(3*t)-math.cos(4*t)))) end; return p,"dot" end
local function GenHexagon() local p={}; for i=0,5 do local a=math.rad(60*i-30); table.insert(p,Vector2.new(math.cos(a)*14, math.sin(a)*14)) end; return p,"line" end
local function GenStar() local p={}; for i=0,9 do local a=math.rad(36*i-90); local r=i%2==0 and 16 or 7; table.insert(p,Vector2.new(math.cos(a)*r, math.sin(a)*r)) end; return p,"line" end
local function GenSquare() local p={}; local s=12; for x=-s,s,2 do table.insert(p,Vector2.new(x,-s)) end; for y=-s,s,2 do table.insert(p,Vector2.new(s,y)) end; for x=s,-s,-2 do table.insert(p,Vector2.new(x,s)) end; for y=s,-s,-2 do table.insert(p,Vector2.new(-s,y)) end; return p,"dot" end
local function GenDot() return {Vector2.new(0,0)}, "dot" end
local function GenPlus() return {{Vector2.new(0,-12),Vector2.new(0,12)},{Vector2.new(-12,0),Vector2.new(12,0)}}, "cross" end

local chPoints, chMode, chDots, chLines, chOutlines = {}, "dot", {}, {}, {}
local function RebuildCrosshair()
    ClearCH(); chDots={}; chLines={}; chOutlines={}
    local gen = ({Heart=GenHeart,Hexagon=GenHexagon,Star=GenStar,Square=GenSquare,Dot=GenDot,Plus=GenPlus})[S.CrosshairType] or GenHeart
    chPoints, chMode = gen()
    if chMode == "dot" then for i=1,#chPoints do local d = NewCHDot(); if d then table.insert(chDots, d) end end
    elseif chMode == "cross" then for i=1,#chPoints do
        if S.CrosshairOutline then local ol = NewCHLine(); if ol then table.insert(chOutlines, ol) end end
        local l = NewCHLine(); if l then table.insert(chLines, l) end end
    else for i=1,#chPoints do if S.CrosshairOutline then local ol = NewCHLine(); if ol then table.insert(chOutlines, ol) end end
        local l = NewCHLine(); if l then table.insert(chLines, l) end end end
end
RebuildCrosshair()

local function ApplyChams(p)
    if not p or not p.Character then return end
    if ChamsObjects[p] then pcall(function() ChamsObjects[p]:Destroy() end) end
    if S.ChamsTeamCheck and IsTeammate(p) then return end
    pcall(function()
        local hl = Instance.new("Highlight")
        hl.FillColor = C3(S.ChamsColor)
        hl.OutlineColor = C3(S.ChamsOutlineColor)
        hl.FillTransparency = S.ChamsFillTransparency/100
        hl.OutlineTransparency = S.ChamsOutlineTransparency/100
        if S.ChamsMode == "Wall" then hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        elseif S.ChamsMode == "Visible" then hl.DepthMode = Enum.HighlightDepthMode.Occluded
        elseif S.ChamsMode == "Invisible" then hl.DepthMode = Enum.HighlightDepthMode.Occluded; hl.FillTransparency = 1 end
        hl.Adornee = p.Character; hl.Parent = p.Character; hl.Name = "AimbotPRO_Chams"
        ChamsObjects[p] = hl
    end)
end

local function RemoveChams(p)
    if ChamsObjects[p] then pcall(function() ChamsObjects[p]:Destroy() end); ChamsObjects[p] = nil end
    if p and p.Character then for _, c in ipairs(p.Character:GetChildren()) do if c.Name == "AimbotPRO_Chams" then pcall(function() c:Destroy() end) end end end
end

local function ApplyGlow(p)
    if not p or not p.Character then return end
    if GlowObjects[p] then pcall(function() GlowObjects[p]:Destroy() end); GlowObjects[p] = nil end
    local r = p.Character:FindFirstChild("HumanoidRootPart")
    if r then pcall(function()
        local l = Instance.new("PointLight"); l.Color = C3(S.GlowColor); l.Brightness = 3; l.Range = 15
        l.Parent = r; l.Name = "AimbotPRO_Glow"; GlowObjects[p] = l
    end) end
end

local function RemoveGlow(p)
    if GlowObjects[p] then pcall(function() GlowObjects[p]:Destroy() end); GlowObjects[p] = nil end
    if p and p.Character then for _, c in ipairs(p.Character:GetChildren()) do if c.Name == "AimbotPRO_Glow" then pcall(function() c:Destroy() end) end end end
end

local function RefreshChams()
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LocalPlayer then continue end
        RemoveChams(p); RemoveGlow(p)
        if S.ChamsEnabled then ApplyChams(p) end
        if S.GlowEnabled then ApplyGlow(p) end
    end
end

local function RefreshGlow()
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LocalPlayer then continue end
        RemoveGlow(p); if S.GlowEnabled then ApplyGlow(p) end
    end
end

-- ═══════════ ANTI-AIM ═══════════
local spinAngle, jitterFlip = 0, 1
local function ApplyAntiAim(dt)
    if not S.AntiAimEnabled then return end
    local ch = LocalPlayer.Character; if not ch then return end
    local hrp = ch:FindFirstChild("HumanoidRootPart"); local hu = ch:FindFirstChildOfClass("Humanoid")
    if not (hrp and hu) then return end
    local yaw, pitch, roll = 0, 0, math.rad(S.RollAngle)
    if S.ManualAA ~= "None" then yaw = math.rad(({Left=90,Right=-90,Back=180,Front=0})[S.ManualAA] or 0)
    elseif S.Freestanding then local cp, cd = nil, math.huge
        for _, p in ipairs(Players:GetPlayers()) do if p == LocalPlayer then continue end
            local c = p.Character; if c and c:FindFirstChild("HumanoidRootPart") then
                local d = (c.HumanoidRootPart.Position - hrp.Position).Magnitude
                if d < cd then cd = d; cp = c end end end
        if cp then local dir = (cp.HumanoidRootPart.Position - hrp.Position).Unit; yaw = math.atan2(-dir.X, -dir.Z) + math.rad(90) end
    else
        if S.YawMode == "Spin" then spinAngle = (spinAngle + dt*S.SpinSpeed*60) % 360; yaw = math.rad(spinAngle)
        elseif S.YawMode == "Backwards" then yaw = math.rad(180)
        elseif S.YawMode == "Jitter" then jitterFlip = -jitterFlip; yaw = math.rad(S.YawJitterRange * jitterFlip)
        elseif S.YawMode == "Random" then yaw = math.rad(math.random(-180, 180))
        elseif S.YawMode == "Custom" then yaw = math.rad(S.YawOffset) end
    end
    yaw = yaw + math.rad(S.YawOffset)
    if S.InverterActive then yaw = yaw + math.rad(180) end
    if S.PitchMode == "Down" then pitch = math.rad(-89)
    elseif S.PitchMode == "Up" then pitch = math.rad(89)
    elseif S.PitchMode == "Fake Down" then pitch = math.rad(-45) end
    pcall(function() hu.AutoRotate = false; hrp.CFrame = CFrame.new(hrp.Position) * CFrame.Angles(pitch, yaw, roll) end)
end

-- Track health
local function TrackPlayerHealth(p)
    if p == LocalPlayer then return end
    task.spawn(function()
        while p.Parent do
            local ch = p.Character
            if ch then local hu = ch:FindFirstChildOfClass("Humanoid")
                if hu then local prev = PreviousHealth[p] or hu.Health; local current = hu.Health
                    if current < prev - 5 and current > 0 and S.HitSoundEnabled then
                        PlaySound(GetSoundID(HitSounds, S.HitSoundID), S.HitSoundVolume/10)
                        if S.DamageText then local head = ch:FindFirstChild("Head")
                            if head then table.insert(DamageTexts, {pos=head.Position+Vector3.new(0,2,0), dmg=math.floor(prev-current), time=tick()}) end end
                    end
                    if prev > 0 and current <= 0 and S.KillSoundEnabled then
                        PlaySound(GetSoundID(KillSounds, S.KillSoundID), S.KillSoundVolume/10)
                        ShowKillMessage(p.Name)
                    end
                    PreviousHealth[p] = current
                end
            end
            task.wait(0.1)
        end
    end)
end

local wjDebounce = false
UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.KeyCode == Enum.KeyCode.Space and S.WallJump and not wjDebounce then
        local ch = LocalPlayer.Character; if not ch then return end
        local hrp = ch:FindFirstChild("HumanoidRootPart"); local hu = ch:FindFirstChildOfClass("Humanoid")
        if hrp and hu and hu:GetState() == Enum.HumanoidStateType.Freefall then
            local params = RaycastParams.new(); params.FilterDescendantsInstances = {ch}
            local result = workspace:Raycast(hrp.Position, hrp.CFrame.LookVector * 3, params)
            if result then wjDebounce = true
                hrp.Velocity = Vector3.new(hrp.Velocity.X, 60, hrp.Velocity.Z) - hrp.CFrame.LookVector * 30
                task.wait(0.3); wjDebounce = false end end
    end
    if not S.AimbotEnabled then return end
    if input.UserInputType == Enum.UserInputType.MouseButton2 or (IS_MOBILE and input.UserInputType == Enum.UserInputType.Touch) then
        if S.AimbotMode == "Hold" then AimbotActive = true
        elseif S.AimbotMode == "Toggle" then ToggleActive = not ToggleActive end
    end
end)

-- ═══════════ MAIN RENDER LOOP ═══════════
local crosshairRot, tt = 0, 0
local fpsFrames, fpsTime, currentFPS = 0, 0, 0
local velTrackTimer, triggerTimer = 0, 0

RunService.RenderStepped:Connect(function(dt)
    tt = tt + dt
    local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    local camPos = Camera.CFrame.Position
    local myChar = LocalPlayer.Character

    ApplyAntiAim(dt)

    -- FPS counter
    fpsFrames = fpsFrames + 1; fpsTime = fpsTime + dt
    if fpsTime >= 0.5 then currentFPS = math.floor(fpsFrames/fpsTime); fpsFrames = 0; fpsTime = 0 end

    -- Velocity tracking
    velTrackTimer = velTrackTimer + dt
    if velTrackTimer >= 0.03 then velTrackTimer = 0; UpdateVelocityTracking() end

    -- Stats HUD
    StatsHUD.Visible = S.ShowFPS or S.ShowMS or S.FakePing
    if S.ShowFPS then FPSLbl.Text = "FPS: "..currentFPS; FPSLbl.Visible = true else FPSLbl.Visible = false end
    if S.ShowMS then MSLbl.Text = "MS: "..math.floor(dt*1000); MSLbl.Visible = true else MSLbl.Visible = false end
    if S.FakePing then PingLbl.Text = "Ping: "..S.FakePingValue.."ms"; PingLbl.Visible = true
    else local ok, p = pcall(function() return math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue()) end)
        PingLbl.Text = ok and "Ping: "..p.."ms" or "Ping: ?"; PingLbl.Visible = S.ShowFPS or S.ShowMS end

    -- Aspect Ratio letterbox
    if S.AspectRatio > 0.1 then
        local vY, vX = Camera.ViewportSize.Y, Camera.ViewportSize.X
        local tH = vX / S.AspectRatio; local bar = math.max((vY - tH)/2, 0)
        LBTop.Size = UDim2.new(1,0,0,bar); LBTop.Visible = bar > 0
        LBBot.Size = UDim2.new(1,0,0,bar); LBBot.Position = UDim2.new(0,0,1,-bar); LBBot.Visible = bar > 0
    else LBTop.Visible = false; LBBot.Visible = false end

    -- Movement
    if myChar then
        local hu = myChar:FindFirstChildOfClass("Humanoid"); local hrp = myChar:FindFirstChild("HumanoidRootPart")
        if hu and hrp then
            if S.BunnyHop and hu:GetState() == Enum.HumanoidStateType.Landed then hu.Jump = true end
            if (S.AutoStrafe or S.StrafeInAir) and hu:GetState() == Enum.HumanoidStateType.Freefall then
                local md = hu.MoveDirection; if md.Magnitude > 0 then hrp.Velocity = Vector3.new(md.X*30, hrp.Velocity.Y, md.Z*30) end end
            if S.QuickStop and hu.MoveDirection.Magnitude < 0.1 then hrp.Velocity = Vector3.new(0, hrp.Velocity.Y, 0) end
            if S.JumpBug and hu:GetState() == Enum.HumanoidStateType.Freefall and hrp.Velocity.Y < -30 then hu:ChangeState(Enum.HumanoidStateType.Landed) end
            if S.LongJump and hu:GetState() == Enum.HumanoidStateType.Freefall then
                local md = hu.MoveDirection; if md.Magnitude > 0 then hrp.Velocity = Vector3.new(md.X*50, hrp.Velocity.Y, md.Z*50) end end
            if S.WalkSpeedBoost then hu.WalkSpeed = S.WalkSpeedValue end
            if S.JumpPowerBoost then hu.JumpPower = S.JumpPowerValue; hu.UseJumpPower = true end
            if S.ExploitSpeed then hu.WalkSpeed = S.ExploitSpeedValue end
            hu.CameraOffset = Vector3.new(S.HandOffsetX, S.HandOffsetY, S.HandOffsetZ)
        end
    end

    -- Remove effects
    if S.RemoveEffects or S.NoParticles then
        for _, obj in ipairs(workspace:GetDescendants()) do
            if S.NoParticles and (obj:IsA("ParticleEmitter") or obj:IsA("Smoke") or obj:IsA("Fire") or obj:IsA("Sparkles")) then
                pcall(function() obj.Enabled = false end) end
        end
    end

    -- Compass
    CompassF.Visible = S.Compass
    if S.Compass and myChar then
        local dir = Camera.CFrame.LookVector; local ang = math.deg(math.atan2(-dir.X, -dir.Z))
        local dirs = {{"N",-22.5,22.5},{"NE",22.5,67.5},{"E",67.5,112.5},{"SE",112.5,157.5},{"SW",-157.5,-112.5},{"W",-112.5,-67.5},{"NW",-67.5,-22.5}}
        local dn = "S"
        for _, d in ipairs(dirs) do if ang > d[2] and ang <= d[3] then dn = d[1]; break end end
        CompassLbl.Text = dn.." ("..math.floor(ang).."°)"
    end

    -- Nearby Enemy
    if S.NearbyEnemy and myChar then
        local myHrp = myChar:FindFirstChild("HumanoidRootPart")
        if myHrp then local near, cd = false, math.huge
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                    local d = (p.Character.HumanoidRootPart.Position - myHrp.Position).Magnitude
                    if d < 30 then near = true; if d < cd then cd = d end end end end
            NearbyF.Visible = near; if near then NearbyLbl.Text = "⚠ ENEMY "..math.floor(cd).."m" end
        end
    else NearbyF.Visible = false end

    -- FOV Circle
    if FOVCircle then FOVCircle.Position = center; FOVCircle.Radius = S.FOV; FOVCircle.Color = C3(S.FOVColor); FOVCircle.Visible = S.ShowFOV end
    if FOVCircleOutline then FOVCircleOutline.Position = center; FOVCircleOutline.Radius = S.FOV; FOVCircleOutline.Visible = S.ShowFOV end

    -- Crosshair
    if S.CrosshairRotation then crosshairRot = (crosshairRot + dt*S.CrosshairRotSpeed) % 360 end
    local rot = math.rad(crosshairRot)
    local szM = S.CrosshairSize/100; local thM = S.CrosshairThickness/100
    local opa = S.CrosshairOpacity/100; local gap = S.CrosshairGap
    local cCol = C3(S.CrosshairColor); local vis = S.ShowCrosshair and opa > 0.01

    if chMode == "dot" then
        for i, d in ipairs(chDots) do
            if chPoints[i] then local pt = chPoints[i]; local sc = Vector2.new(pt.X*szM, pt.Y*szM)
                if gap > 0 and sc.Magnitude > 0 then sc = sc + sc.Unit*gap end
                d.Position = center + RotateV2(sc, rot)
                d.Radius = S.CrosshairType == "Dot" and S.CrosshairDotSize*thM or 1.6*thM
                d.Color = cCol; d.Transparency = opa; d.Visible = vis
            else d.Visible = false end end
        for _, l in ipairs(chLines) do l.Visible = false end
        for _, l in ipairs(chOutlines) do l.Visible = false end
    elseif chMode == "cross" then
        for _, d in ipairs(chDots) do d.Visible = false end
        for i, l in ipairs(chLines) do
            if chPoints[i] then
                local p1 = Vector2.new(chPoints[i][1].X*szM, chPoints[i][1].Y*szM)
                local p2 = Vector2.new(chPoints[i][2].X*szM, chPoints[i][2].Y*szM)
                l.From = center + RotateV2(p1, rot); l.To = center + RotateV2(p2, rot)
                l.Color = cCol; l.Thickness = 2*thM; l.Transparency = opa; l.Visible = vis
            else l.Visible = false end end
    else
        for _, d in ipairs(chDots) do d.Visible = false end
        for i, l in ipairs(chLines) do
            if chPoints[i] then local ni = i%#chPoints + 1; local p1, p2 = chPoints[i], chPoints[ni]
                local s1 = Vector2.new(p1.X*szM, p1.Y*szM); local s2 = Vector2.new(p2.X*szM, p2.Y*szM)
                if gap > 0 then if s1.Magnitude > 0 then s1 = s1+s1.Unit*gap end; if s2.Magnitude > 0 then s2 = s2+s2.Unit*gap end end
                l.From = center + RotateV2(s1, rot); l.To = center + RotateV2(s2, rot)
                l.Color = cCol; l.Thickness = 2*thM; l.Transparency = opa; l.Visible = vis
            else l.Visible = false end end
        for i, ol in ipairs(chOutlines) do
            if S.CrosshairOutline and chLines[i] and chLines[i].Visible then
                ol.From = chLines[i].From; ol.To = chLines[i].To
                ol.Color = C3(S.CrosshairOutlineColor); ol.Thickness = 2*thM + 2
                ol.Transparency = opa*0.7; ol.Visible = vis
            else if ol then ol.Visible = false end end end
    end

    -- AIMBOT
    local shouldAim = false
    if S.AimbotEnabled then
        if S.AimbotMode == "Always" then shouldAim = true
        elseif S.AimbotMode == "Hold" then shouldAim = AimbotActive
        elseif S.AimbotMode == "Toggle" then shouldAim = ToggleActive end
    end
    if S.RageEnabled and S.RageAutoFire then shouldAim = true end

    if shouldAim then
        local target, aimPos = GetBestTarget()
        if target and aimPos then
            if S.AimShake > 0 then local sh = S.AimShake * 0.01
                aimPos = aimPos + Vector3.new((math.random()-0.5)*sh, (math.random()-0.5)*sh, (math.random()-0.5)*sh) end
            local targetCF = CFrame.new(camPos, aimPos)
            local baseLerp = dt * (22 - S.Smoothness) * 3
            if S.AimAcceleration then local sp, _ = Camera:WorldToViewportPoint(aimPos)
                local dist = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                local accel = math.clamp(dist / S.FOV, 0.1, 1) * S.AccelFactor
                baseLerp = baseLerp * accel end
            if S.RageEnabled then baseLerp = 1 end -- Instant for rage
            Camera.CFrame = Camera.CFrame:Lerp(targetCF, math.clamp(baseLerp, 0, 1))

            -- Rage auto fire
            if S.RageEnabled and S.RageAutoFire then FireTriggerbot() end
        end
    end

    -- TRIGGERBOT
    if S.TriggerbotEnabled then
        triggerTimer = triggerTimer + dt
        local canTrigger = true
        if S.TriggerbotMode == "Hold Key" then canTrigger = UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) end

        if canTrigger then
            local tTarget, tPart = GetTriggerbotTarget()
            if tTarget and tPart then
                local delay = S.TriggerbotDelay
                if S.TriggerbotRandomDelay then delay = math.random(S.TriggerbotRandomMin, S.TriggerbotRandomMax) end

                if triggerTimer*1000 >= delay and not triggerCooldown then
                    triggerCooldown = true
                    task.spawn(function()
                        if S.TriggerbotBurst then
                            for i = 1, S.TriggerbotBurstCount do
                                FireTriggerbot()
                                task.wait(S.TriggerbotBurstDelay/1000)
                            end
                        else
                            FireTriggerbot()
                            task.wait(S.TriggerbotHoldTime/1000)
                        end
                        triggerCooldown = false; triggerTimer = 0
                    end)
                end
            else
                triggerTimer = 0
            end
        end
    end

    -- Radar
    RadarF.Visible = S.RadarEnabled
    if S.RadarEnabled then
        local lp = myChar and myChar:FindFirstChild("HumanoidRootPart")
        for p, dot in pairs(RadarDots) do
            if not p.Parent or not p.Character then dot.Visible = false
            else local root = p.Character:FindFirstChild("HumanoidRootPart")
                if root and lp then local rel = root.Position - lp.Position
                    local look = Vector3.new(Camera.CFrame.LookVector.X, 0, Camera.CFrame.LookVector.Z)
                    if look.Magnitude > 0 then look = look.Unit end
                    local ang = math.atan2(look.X, look.Z); local cs, sn = math.cos(-ang), math.sin(-ang)
                    local rx = rel.X*cs - rel.Z*sn; local rz = rel.X*sn + rel.Z*cs
                    dot.Position = UDim2.new(math.clamp(0.5+rx/S.RadarZoom/2,0,1),-2, math.clamp(0.5+rz/S.RadarZoom/2,0,1),-2)
                    dot.BackgroundColor3 = GetESPColor(p); dot.Visible = true
                else dot.Visible = false end end end
        for _, p in ipairs(Players:GetPlayers()) do if p ~= LocalPlayer then GetRadarDot(p) end end
    else for _, d in pairs(RadarDots) do d.Visible = false end end

    -- ═══════════ ESP RENDERING ═══════════
    for player, esp in pairs(ESPCache) do
        if not player or not player.Parent then HideAllESP(esp); continue end
        if not S.ESPEnabled then HideAllESP(esp); continue end
        local ch = player.Character; if not ch then HideAllESP(esp); continue end
        local hu = ch:FindFirstChildOfClass("Humanoid"); local root = ch:FindFirstChild("HumanoidRootPart"); local head = ch:FindFirstChild("Head")
        if not (hu and hu.Health > 0 and root and head) then HideAllESP(esp); continue end
        local wD = (root.Position - camPos).Magnitude; if wD > S.ESPMaxDist then HideAllESP(esp); continue end
        local ht = head.Position + Vector3.new(0, head.Size.Y/2 + 0.3, 0)
        local rb = root.Position - Vector3.new(0, 3, 0)
        local hs, hv = Camera:WorldToViewportPoint(ht); local rs, rv = Camera:WorldToViewportPoint(rb)
        if not (hv and rv) then HideAllESP(esp); continue end
        local top, bot = hs.Y, rs.Y; local h = math.max(bot - top, 10)
        local cx = (hs.X + rs.X) / 2; local w = math.max(h * 0.5, 15)
        local espCol = GetESPColor(player)

        -- Reset all visibilities
        for _, l in ipairs(esp.BoxLines) do l.Visible = false end
        for _, l in ipairs(esp.BoxOutlines) do l.Visible = false end
        for _, l in ipairs(esp.BoxOutlines2) do l.Visible = false end
        for _, l in ipairs(esp.CornerLines) do l.Visible = false end
        for _, l in ipairs(esp.CornerOutlines) do l.Visible = false end
        for _, l in ipairs(esp.Box3D) do l.Visible = false end
        for _, l in ipairs(esp.Box3DOutlines) do l.Visible = false end
        if esp.FillQuad then esp.FillQuad.Visible = false end

        -- BOXES
        if S.ESPBoxes then
            if S.ESPBoxStyle == "2D" then
                local tl = Vector2.new(cx-w/2, top); local tr2 = Vector2.new(cx+w/2, top)
                local bl = Vector2.new(cx-w/2, bot); local br = Vector2.new(cx+w/2, bot)
                local pts = {tl, tr2, br, bl}
                for i = 1, 4 do local ni = i%4 + 1
                    if S.ESPOutline then
                        if esp.BoxOutlines2[i] then esp.BoxOutlines2[i].From = pts[i]; esp.BoxOutlines2[i].To = pts[ni]; esp.BoxOutlines2[i].Thickness = S.ESPThickness + 4; esp.BoxOutlines2[i].Visible = true end
                        if esp.BoxOutlines[i] then esp.BoxOutlines[i].From = pts[i]; esp.BoxOutlines[i].To = pts[ni]; esp.BoxOutlines[i].Thickness = S.ESPThickness + 2; esp.BoxOutlines[i].Visible = true end
                    end
                    if esp.BoxLines[i] then esp.BoxLines[i].From = pts[i]; esp.BoxLines[i].To = pts[ni]; esp.BoxLines[i].Color = espCol; esp.BoxLines[i].Thickness = S.ESPThickness; esp.BoxLines[i].Visible = true end end
                if S.ESPFillBox and esp.FillQuad then esp.FillQuad.PointA = tl; esp.FillQuad.PointB = tr2; esp.FillQuad.PointC = br; esp.FillQuad.PointD = bl; esp.FillQuad.Color = espCol; esp.FillQuad.Transparency = S.ESPFillOpacity/100; esp.FillQuad.Visible = true end
            elseif S.ESPBoxStyle == "Corner" then
                local cl = math.min(w, h) * 0.25
                local corners = {
                    {Vector2.new(cx-w/2, top), Vector2.new(cx-w/2+cl, top)},
                    {Vector2.new(cx-w/2, top), Vector2.new(cx-w/2, top+cl)},
                    {Vector2.new(cx+w/2, top), Vector2.new(cx+w/2-cl, top)},
                    {Vector2.new(cx+w/2, top), Vector2.new(cx+w/2, top+cl)},
                    {Vector2.new(cx-w/2, bot), Vector2.new(cx-w/2+cl, bot)},
                    {Vector2.new(cx-w/2, bot), Vector2.new(cx-w/2, bot-cl)},
                    {Vector2.new(cx+w/2, bot), Vector2.new(cx+w/2-cl, bot)},
                    {Vector2.new(cx+w/2, bot), Vector2.new(cx+w/2, bot-cl)}
                }
                for i = 1, 8 do
                    if S.ESPOutline and esp.CornerOutlines[i] then esp.CornerOutlines[i].From = corners[i][1]; esp.CornerOutlines[i].To = corners[i][2]; esp.CornerOutlines[i].Thickness = 4.5; esp.CornerOutlines[i].Visible = true end
                    if esp.CornerLines[i] then esp.CornerLines[i].From = corners[i][1]; esp.CornerLines[i].To = corners[i][2]; esp.CornerLines[i].Color = espCol; esp.CornerLines[i].Thickness = 2.5; esp.CornerLines[i].Visible = true end end
            elseif S.ESPBoxStyle == "3D" then
                local ext = Vector3.new(2, 3, 1.5)
                local c3d = {root.Position+Vector3.new(-ext.X,ext.Y,-ext.Z),root.Position+Vector3.new(ext.X,ext.Y,-ext.Z),root.Position+Vector3.new(ext.X,ext.Y,ext.Z),root.Position+Vector3.new(-ext.X,ext.Y,ext.Z),root.Position+Vector3.new(-ext.X,-ext.Y,-ext.Z),root.Position+Vector3.new(ext.X,-ext.Y,-ext.Z),root.Position+Vector3.new(ext.X,-ext.Y,ext.Z),root.Position+Vector3.new(-ext.X,-ext.Y,ext.Z)}
                local sp3d = {}; local allOn = true
                for i, c in ipairs(c3d) do local sp, on = Camera:WorldToViewportPoint(c); sp3d[i] = Vector2.new(sp.X, sp.Y); if not on then allOn = false end end
                if allOn then local edges = {{1,2},{2,3},{3,4},{4,1},{5,6},{6,7},{7,8},{8,5},{1,5},{2,6},{3,7},{4,8}}
                    for i, e in ipairs(edges) do
                        if S.ESPOutline and esp.Box3DOutlines[i] then esp.Box3DOutlines[i].From = sp3d[e[1]]; esp.Box3DOutlines[i].To = sp3d[e[2]]; esp.Box3DOutlines[i].Thickness = 3; esp.Box3DOutlines[i].Visible = true end
                        if esp.Box3D[i] then esp.Box3D[i].From = sp3d[e[1]]; esp.Box3D[i].To = sp3d[e[2]]; esp.Box3D[i].Color = espCol; esp.Box3D[i].Thickness = 1.5; esp.Box3D[i].Visible = true end end end
            end
        end

        -- SKELETON
        for _, l in ipairs(esp.Skeleton) do l.Visible = false end
        for _, l in ipairs(esp.SkeletonOut) do l.Visible = false end
        if S.ESPSkeleton then
            for i, bone in ipairs(R15Bones) do
                local p1 = ch:FindFirstChild(bone[1]); local p2 = ch:FindFirstChild(bone[2])
                if p1 and p2 and esp.Skeleton[i] then
                    local s1, o1 = Camera:WorldToViewportPoint(p1.Position)
                    local s2, o2 = Camera:WorldToViewportPoint(p2.Position)
                    if o1 and o2 then
                        if S.ESPOutline and esp.SkeletonOut[i] then
                            esp.SkeletonOut[i].From = Vector2.new(s1.X, s1.Y); esp.SkeletonOut[i].To = Vector2.new(s2.X, s2.Y)
                            esp.SkeletonOut[i].Thickness = S.ESPSkeletonThickness + 2; esp.SkeletonOut[i].Visible = true end
                        esp.Skeleton[i].From = Vector2.new(s1.X, s1.Y); esp.Skeleton[i].To = Vector2.new(s2.X, s2.Y)
                        esp.Skeleton[i].Color = C3(S.SkeletonColor); esp.Skeleton[i].Thickness = S.ESPSkeletonThickness; esp.Skeleton[i].Visible = true
                    end
                end
            end
        end

        -- NAMES
        if S.ESPNames and esp.NameText then
            esp.NameText.Text = player.Name; esp.NameText.Position = Vector2.new(cx, top - 16)
            esp.NameText.Color = espCol; esp.NameText.Size = S.ESPTextSize; esp.NameText.Visible = true
        elseif esp.NameText then esp.NameText.Visible = false end

        -- HEALTH BAR
        if esp.HealthBG then esp.HealthBG.Visible = false end
        if esp.HealthBGOutline then esp.HealthBGOutline.Visible = false end
        if esp.HealthFill then esp.HealthFill.Visible = false end
        if esp.HealthText then esp.HealthText.Visible = false end
        if S.ESPHealth then
            local hp = math.clamp(hu.Health / hu.MaxHealth, 0, 1)
            local hpCol = Color3.fromRGB(math.floor((1-hp)*220), math.floor(hp*220), 30)
            local bx = cx - w/2 - 7
            if S.ESPBarStyle == "Right" then bx = cx + w/2 + 7 end
            if esp.HealthBGOutline then esp.HealthBGOutline.From = Vector2.new(bx, top-1); esp.HealthBGOutline.To = Vector2.new(bx, bot+1); esp.HealthBGOutline.Visible = true end
            if esp.HealthBG then esp.HealthBG.From = Vector2.new(bx, top); esp.HealthBG.To = Vector2.new(bx, bot); esp.HealthBG.Visible = true end
            if esp.HealthFill then esp.HealthFill.Color = hpCol; esp.HealthFill.From = Vector2.new(bx, bot); esp.HealthFill.To = Vector2.new(bx, bot-h*hp); esp.HealthFill.Visible = true end
            if S.ESPHealthNumber and esp.HealthText then
                esp.HealthText.Text = math.floor(hu.Health); esp.HealthText.Position = Vector2.new(bx, bot-h*hp-10)
                esp.HealthText.Color = hpCol; esp.HealthText.Visible = hp < 1
            end
        end

        -- DISTANCE
        if S.ESPDistance and esp.DistText then
            esp.DistText.Text = "["..math.floor(wD).."m]"; esp.DistText.Position = Vector2.new(cx, bot + 4)
            esp.DistText.Color = espCol; esp.DistText.Visible = true
        elseif esp.DistText then esp.DistText.Visible = false end

        -- WEAPON
        if S.ESPWeapon and esp.WeaponText then
            local tool = ch:FindFirstChildOfClass("Tool")
            if tool then esp.WeaponText.Text = "["..tool.Name.."]"; esp.WeaponText.Position = Vector2.new(cx, bot + 18); esp.WeaponText.Visible = true
            else esp.WeaponText.Visible = false end
        elseif esp.WeaponText then esp.WeaponText.Visible = false end

        -- FLAGS
        if S.ESPFlags and esp.FlagsText then
            local flags = {}
            if hu:GetState() == Enum.HumanoidStateType.Freefall then table.insert(flags, "AIR") end
            if hu.Health < hu.MaxHealth * 0.3 then table.insert(flags, "LOW") end
            local tool = ch:FindFirstChildOfClass("Tool"); if tool then table.insert(flags, tool.Name) end
            if #flags > 0 then
                esp.FlagsText.Text = table.concat(flags, " | ")
                esp.FlagsText.Position = Vector2.new(cx + w/2 + 6, top)
                esp.FlagsText.Center = false; esp.FlagsText.Visible = true
            else esp.FlagsText.Visible = false end
        elseif esp.FlagsText then esp.FlagsText.Visible = false end

        -- TRACERS
        if S.ESPTracers then
            local tracerFrom
            if S.ESPTracerOrigin == "Bottom" then tracerFrom = Vector2.new(center.X, Camera.ViewportSize.Y)
            elseif S.ESPTracerOrigin == "Top" then tracerFrom = Vector2.new(center.X, 0)
            elseif S.ESPTracerOrigin == "Center" then tracerFrom = center
            elseif S.ESPTracerOrigin == "Mouse" then tracerFrom = Vector2.new(Mouse.X, Mouse.Y) end
            if esp.TracerOutline and S.ESPOutline then esp.TracerOutline.From = tracerFrom; esp.TracerOutline.To = Vector2.new(cx, bot); esp.TracerOutline.Visible = true end
            if esp.Tracer then esp.Tracer.From = tracerFrom; esp.Tracer.To = Vector2.new(cx, bot); esp.Tracer.Color = espCol; esp.Tracer.Visible = true end
        else if esp.Tracer then esp.Tracer.Visible = false end; if esp.TracerOutline then esp.TracerOutline.Visible = false end end

        -- SNAP LINES
        if S.ESPSnapLines and esp.SnapLine then
            esp.SnapLine.From = center; esp.SnapLine.To = Vector2.new(cx, (top+bot)/2)
            esp.SnapLine.Color = espCol; esp.SnapLine.Transparency = 0.5; esp.SnapLine.Visible = true
        elseif esp.SnapLine then esp.SnapLine.Visible = false end

        -- HEAD DOT
        if S.ESPHeadDot then
            local headSP, headOn = Camera:WorldToViewportPoint(head.Position)
            if headOn then
                local hp2 = Vector2.new(headSP.X, headSP.Y)
                if S.ESPOutline then
                    if esp.HeadDotOutline2 then esp.HeadDotOutline2.Position = hp2; esp.HeadDotOutline2.Radius = S.ESPHeadDotSize+3; esp.HeadDotOutline2.Visible = true end
                    if esp.HeadDotOutline then esp.HeadDotOutline.Position = hp2; esp.HeadDotOutline.Radius = S.ESPHeadDotSize+1.5; esp.HeadDotOutline.Visible = true end
                end
                if esp.HeadDot then esp.HeadDot.Position = hp2; esp.HeadDot.Radius = S.ESPHeadDotSize; esp.HeadDot.Color = espCol; esp.HeadDot.Filled = true; esp.HeadDot.Visible = true end
            else
                if esp.HeadDot then esp.HeadDot.Visible = false end
                if esp.HeadDotOutline then esp.HeadDotOutline.Visible = false end
                if esp.HeadDotOutline2 then esp.HeadDotOutline2.Visible = false end
            end
        else
            if esp.HeadDot then esp.HeadDot.Visible = false end
            if esp.HeadDotOutline then esp.HeadDotOutline.Visible = false end
            if esp.HeadDotOutline2 then esp.HeadDotOutline2.Visible = false end
        end
    end

    -- Damage Text
    for i = #DamageTexts, 1, -1 do
        local d = DamageTexts[i]; local elapsed = tick() - d.time
        if elapsed > 1.5 then
            if DmgDrawings[i] then pcall(function() DmgDrawings[i]:Remove() end); DmgDrawings[i] = nil end
            table.remove(DamageTexts, i)
        else
            local pos = d.pos + Vector3.new(0, elapsed*3, 0)
            local sp, on = Camera:WorldToViewportPoint(pos)
            if on then
                if not DmgDrawings[i] then DmgDrawings[i] = MT(18, C3(S.DamageTextColor), true) end
                if DmgDrawings[i] then
                    DmgDrawings[i].Text = "-"..d.dmg; DmgDrawings[i].Position = Vector2.new(sp.X, sp.Y)
                    DmgDrawings[i].Transparency = math.clamp(1-elapsed/1.5,0,1)
                    DmgDrawings[i].Color = C3(S.DamageTextColor); DmgDrawings[i].Visible = true
                end
            else if DmgDrawings[i] then DmgDrawings[i].Visible = false end end
        end
    end
end)
