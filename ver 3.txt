-- ========== CROSSHAIR DRAWING SYSTEM ==========
local CrosshairDrawings = {}
local MAX_CROSSHAIR_DRAWINGS = 200

local function ClearCrosshairDrawings()
    for _, d in ipairs(CrosshairDrawings) do
        pcall(function() d:Remove() end)
    end
    CrosshairDrawings = {}
end

local function NewCrosshairDot()
    local s, d = pcall(Drawing.new, "Circle")
    if s and d then
        d.Filled = true
        d.Visible = false
        d.Transparency = 1
        table.insert(CrosshairDrawings, d)
        return d
    end
    return nil
end

local function NewCrosshairLine()
    local s, l = pcall(Drawing.new, "Line")
    if s and l then
        l.Visible = false
        l.Transparency = 1
        table.insert(CrosshairDrawings, l)
        return l
    end
    return nil
end

-- Generate shape points
local function GetHeartPoints()
    local pts = {}
    for t = 0, 2 * math.pi, 0.15 do
        local x = 14 * (math.sin(t) ^ 3)
        local y = -(11 * math.cos(t) - 4 * math.cos(2 * t) - 2 * math.cos(3 * t) - math.cos(4 * t))
        table.insert(pts, Vector2.new(x, y))
    end
    return pts
end

local function GetHexagonPoints()
    local pts = {}
    for i = 0, 5 do
        local angle = math.rad(60 * i - 30)
        table.insert(pts, Vector2.new(math.cos(angle) * 14, math.sin(angle) * 14))
    end
    return pts
end

local function GetStarPoints()
    local pts = {}
    for i = 0, 9 do
        local angle = math.rad(36 * i - 90)
        local r = (i % 2 == 0) and 16 or 7
        table.insert(pts, Vector2.new(math.cos(angle) * r, math.sin(angle) * r))
    end
    return pts
end

local function GetSquarePoints()
    local pts = {}
    local s = 12
    for x = -s, s, 3 do table.insert(pts, Vector2.new(x, -s)) end
    for y = -s, s, 3 do table.insert(pts, Vector2.new(s, y)) end
    for x = s, -s, -3 do table.insert(pts, Vector2.new(x, s)) end
    for y = s, -s, -3 do table.insert(pts, Vector2.new(-s, y)) end
    return pts
end

local function GetDotPoints()
    return {Vector2.new(0, 0)}
end

local function GetCrosshairPoints(crossType)
    if crossType == "Heart" then return GetHeartPoints()
    elseif crossType == "Hexagon" then return GetHexagonPoints()
    elseif crossType == "Star" then return GetStarPoints()
    elseif crossType == "Square" then return GetSquarePoints()
    elseif crossType == "Dot" then return GetDotPoints()
    end
    return GetHeartPoints()
end

-- Build crosshair drawings
local crosshairPoints = {}
local crosshairDots = {}
local crosshairLines = {}
local crosshairOutlineLines = {}

local function RebuildCrosshair()
    ClearCrosshairDrawings()
    crosshairDots = {}
    crosshairLines = {}
    crosshairOutlineLines = {}

    local cType = Settings.CrosshairType
    crosshairPoints = GetCrosshairPoints(cType)

    if cType == "Dot" then
        local d = NewCrosshairDot()
        if d then table.insert(crosshairDots, d) end
    elseif cType == "Heart" then
        for i = 1, #crosshairPoints do
            local d = NewCrosshairDot()
            if d then table.insert(crosshairDots, d) end
        end
    elseif cType == "Hexagon" or cType == "Star" or cType == "Square" then
        if cType == "Hexagon" or cType == "Star" then
            for i = 1, #crosshairPoints do
                if Settings.CrosshairOutline then
                    local ol = NewCrosshairLine()
                    if ol then table.insert(crosshairOutlineLines, ol) end
                end
                local l = NewCrosshairLine()
                if l then table.insert(crosshairLines, l) end
            end
        else
            for i = 1, #crosshairPoints do
                local d = NewCrosshairDot()
                if d then table.insert(crosshairDots, d) end
            end
        end
    end
end

local ESPData = {}

local function NewLine()
    local s, l = pcall(Drawing.new, "Line")
    if s and l then
        l.Color = C3(Settings.ESPColor)
        l.Thickness = 1.5
        l.Transparency = 1

local function NewText(size)
    local s, t = pcall(Drawing.new, "Text")
    if s and t then
        t.Color = C3(Settings.ESPColor)
        t.Size = size or 13
        t.Font = Drawing.Fonts.UI
        t.Outline = true
        t.OutlineColor = Color3.new(0, 0, 0)
        t.Visible = false
        t.Center = true
        return t
    end
end

local function CreateESP(player)
    if ESPData[player] then return end
    local esp = {
        BoxLines = {},
        CornerLines = {},
        Box3DLines = {},
        SkeletonLines = {},
    }
    for i = 1, 4 do table.insert(esp.BoxLines, NewLine()) end
    for i = 1, 8 do table.insert(esp.CornerLines, NewLine()) end
    for i = 1, 12 do table.insert(esp.Box3DLines, NewLine()) end
    for i = 1, 6 do table.insert(esp.SkeletonLines, NewLine()) end

    esp.NameText = NewText(13)
    esp.DistText = NewText(11)
    esp.WeaponText = NewText(11)
    esp.HpBG = NewLine()
    if esp.HpBG then
        esp.HpBG.Color = Color3.new(0, 0, 0)
        esp.HpBG.Thickness = 5
        esp.HpBG.Transparency = 0.6
    end
    esp.HpFill = NewLine()
    if esp.HpFill then esp.HpFill.Thickness = 3 end
    esp.Tracer = NewLine()
    esp.SnapLine = NewLine()
    esp.DormantMarker = NewLine()

    ESPData[player] = esp
end

local function RemoveESP(player)
    local esp = ESPData[player]
    if not esp then return end
    for _, group in pairs({esp.BoxLines, esp.CornerLines, esp.Box3DLines, esp.SkeletonLines}) do
        for _, l in ipairs(group) do
            if l then pcall(function() l:Remove() end) end
        end
    end
    for _, k in ipairs({"NameText", "DistText", "WeaponText", "HpBG", "HpFill", "Tracer", "SnapLine", "DormantMarker"}) do
        if esp[k] then pcall(function() esp[k]:Remove() end) end
    end
    ESPData[player] = nil
end

local function HideESP(esp)
    if not esp then return end
    for _, group in pairs({esp.BoxLines, esp.CornerLines, esp.Box3DLines, esp.SkeletonLines}) do
        for _, l in ipairs(group) do
            if l then l.Visible = false end
        end
    end
    for _, k in ipairs({"NameText", "DistText", "WeaponText", "HpBG", "HpFill", "Tracer", "SnapLine", "DormantMarker"}) do
        if esp[k] then esp[k].Visible = false end
    end
end

-- ========== CHAMS ==========
local ChamsObjects = {}
local GlowObjects = {}

local function ApplyChams(player)
    if not player.Character then return end
    if ChamsObjects[player] then pcall(function() ChamsObjects[player]:Destroy() end) end
    local s, h = pcall(function()
        local hl = Instance.new("Highlight")
        hl.FillColor = C3(Settings.ChamsColor)
        hl.OutlineColor = C3(Settings.ChamsOutlineColor)
        hl.FillTransparency = 0.45
        hl.OutlineTransparency = 0
        if Settings.ChamsMode == "Visible" then
            hl.DepthMode = Enum.HighlightDepthMode.Occluded
        elseif Settings.ChamsMode == "Invisible" then
            hl.DepthMode = Enum.HighlightDepthMode.Occluded
            hl.FillTransparency = 1
        elseif Settings.ChamsMode == "Wall" then
            hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        end
        hl.Adornee = player.Character
        hl.Parent = player.Character
        return hl
    end)
    if s and h then ChamsObjects[player] = h end
end

local function RemoveChams(player)
    if ChamsObjects[player] then
        pcall(function() ChamsObjects[player]:Destroy() end)
        ChamsObjects[player] = nil
    end
end

local function ApplyGlow(player)
    if GlowObjects[player] then pcall(function() GlowObjects[player]:Destroy() end); GlowObjects[player] = nil end
    if not player.Character then return end
    local root = player.Character:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local s, g = pcall(function()
        local light = Instance.new("PointLight")
        light.Color = C3(Settings.GlowColor)
        light.Brightness = 3
        light.Range = 15
        light.Parent = root
        return light
    end)
    if s and g then GlowObjects[player] = g end
end

local function RemoveGlow(player)
    if GlowObjects[player] then
        pcall(function() GlowObjects[player]:Destroy() end)
        GlowObjects[player] = nil
    end
end

local function RefreshChams()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            RemoveChams(p)
            RemoveGlow(p)
            if Settings.ChamsEnabled then ApplyChams(p) end
            if Settings.GlowEnabled then ApplyGlow(p) end
        end
    end
end

local function UpdateChamsColor()
    for _, h in pairs(ChamsObjects) do
        pcall(function()
            h.FillColor = C3(Settings.ChamsColor)
            h.OutlineColor = C3(Settings.ChamsOutlineColor)
        end)
    end
    for _, g in pairs(GlowObjects) do
        pcall(function() g.Color = C3(Settings.GlowColor) end)
    end
end

-- ========== WALL CHECK ==========
local function CanSee(part)
    if not Settings.WallCheck then return true end
    local origin = Camera.CFrame.Position
    local dir = (part.Position - origin).Unit * 1000
    local ray = Ray.new(origin, dir)
    local ignore = {workspace.Terrain, LocalPlayer.Character}
    local hit = workspace:FindPartOnRayWithIgnoreList(ray, ignore)
    if hit then
        local ch = part:FindFirstAncestorWhichIsA("Model")
        if ch and ch:IsAncestorOf(hit) then return true end
        return false
    end
    return true
end

-- ========== HITBOX ==========
local HitboxMap = {
    Head = "Head",
    Chest = "UpperTorso",
    Stomach = "LowerTorso",
    Pelvis = "HumanoidRootPart",
    Body = "HumanoidRootPart"
}

local function GetHitboxPart(character, hitboxName)
    local pn = HitboxMap[hitboxName] or "Head"
    return character:FindFirstChild(pn) or character:FindFirstChild("Torso") or character:FindFirstChild("UpperTorso") or character:FindFirstChild("HumanoidRootPart")
end

local function GetMultipoint(part)
    if not Settings.AimbotMultipoint or not part then return {part.Position} end
    local sz = part.Size
    local sc = Settings.AimbotMultipointScale / 100
    local cf = part.CFrame
    return {
        cf.Position,
        (cf * CFrame.new((sz.X / 2) * sc, 0, 0)).Position,
        (cf * CFrame.new(-(sz.X / 2) * sc, 0, 0)).Position,
        (cf * CFrame.new(0, (sz.Y / 2) * sc, 0)).Position,
        (cf * CFrame.new(0, -(sz.Y / 2) * sc, 0)).Position,
    }
end

local function GetClosestPlayer()
    local closest, closestDist, closestPos = nil, Settings.FOV, nil
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local useHitbox = Settings.AimbotHitbox
    if Settings.BaimEnabled then
        local ch = LocalPlayer.Character
        local hu = ch and ch:FindFirstChildOfClass("Humanoid")
        local air = hu and (hu:GetState() == Enum.HumanoidStateType.Freefall or hu:GetState() == Enum.HumanoidStateType.Jumping)
        if Settings.BaimInAir and air then useHitbox = "Chest" end
        if Settings.BaimAfterMisses > 0 and MissCount >= Settings.BaimAfterMisses then useHitbox = "Chest" end
    end
    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then continue end
        if Settings.TeamCheck and player.Team == LocalPlayer.Team then continue end
        local ch = player.Character
        if not ch then continue end
        local hu = ch:FindFirstChildOfClass("Humanoid")
        local pt = GetHitboxPart(ch, useHitbox)
        if hu and hu.Health > 0 and pt then
            for _, pos in ipairs(GetMultipoint(pt)) do
                local sp, os = Camera:WorldToViewportPoint(pos)
                if os then
                    local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                    if d < closestDist and CanSee(pt) then
                        closestDist = d
                        closest = player
                        closestPos = pos
                    end
                end
            end
        end
    end
    return closest, closestPos
end

-- ========== ANTI-AIM ==========
local spinAngle = 0
local jitterFlip = 1

local function GetClosestEnemyDir()
    local ch = LocalPlayer.Character
    if not ch then return nil end
    local hrp = ch:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local cp, cd = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LocalPlayer then continue end
        local c = p.Character
        if c and c:FindFirstChild("HumanoidRootPart") then
            local d = (c.HumanoidRootPart.Position - hrp.Position).Magnitude
            if d < cd then cd = d; cp = c end
        end
    end
    if cp then return (cp.HumanoidRootPart.Position - hrp.Position).Unit end
    return nil
end

local function ApplyAntiAim(dt)
    if not Settings.AntiAimEnabled then return end
    local ch = LocalPlayer.Character
    if not ch then return end
    local hrp = ch:FindFirstChild("HumanoidRootPart")
    local hu = ch:FindFirstChildOfClass("Humanoid")
    if not hrp or not hu then return end
    local yaw, pitch, roll = 0, 0, math.rad(Settings.RollAngle)

    if Settings.ManualAA ~= "None" then
        if Settings.ManualAA == "Left" then yaw = math.rad(90)
        elseif Settings.ManualAA == "Right" then yaw = math.rad(-90)
        elseif Settings.ManualAA == "Back" then yaw = math.rad(180)
        elseif Settings.ManualAA == "Front" then yaw = 0 end
    elseif Settings.Freestanding then
        local d = GetClosestEnemyDir()
        if d then yaw = math.atan2(-d.X, -d.Z) + math.rad(90) end
    else
        if Settings.YawMode == "Spin" then
            spinAngle = (spinAngle + dt * Settings.SpinSpeed * 60) % 360
            yaw = math.rad(spinAngle)
        elseif Settings.YawMode == "Backwards" then yaw = math.rad(180)
        elseif Settings.YawMode == "Jitter" then
            jitterFlip = -jitterFlip
            yaw = math.rad(Settings.YawJitterRange * jitterFlip)
        elseif Settings.YawMode == "Random" then yaw = math.rad(math.random(-180, 180))
        elseif Settings.YawMode == "Custom" then yaw = math.rad(Settings.YawOffset)
        end
    end
    yaw = yaw + math.rad(Settings.YawOffset)
    if Settings.InverterActive then yaw = yaw + math.rad(180) end

    if Settings.PitchMode == "Down" then pitch = math.rad(-89)
    elseif Settings.PitchMode == "Up" then pitch = math.rad(89)
    elseif Settings.PitchMode == "Zero" then pitch = 0
    elseif Settings.PitchMode == "Fake Down" then pitch = math.rad(-45)
    end

    pcall(function()
        hu.AutoRotate = false
        hrp.CFrame = CFrame.new(hrp.Position) * CFrame.Angles(pitch, yaw, roll)
    end)
end

-- ========== MOVEMENT ==========
local wallJumpDebounce = false

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.KeyCode == Enum.KeyCode.Space and Settings.WallJump and not wallJumpDebounce then
        local ch = LocalPlayer.Character
        if not ch then return end
        local hrp = ch:FindFirstChild("HumanoidRootPart")
        local hum = ch:FindFirstChildOfClass("Humanoid")
        if not (hrp and hum) then return end
        if hum:GetState() == Enum.HumanoidStateType.Freefall then
            local ray = Ray.new(hrp.Position, hrp.CFrame.LookVector * 3)
            local hit = workspace:FindPartOnRayWithIgnoreList(ray, {ch})
            if hit then
                wallJumpDebounce = true
                hrp.Velocity = Vector3.new(hrp.Velocity.X, 60, hrp.Velocity.Z) - (hrp.CFrame.LookVector * 30)
                task.wait(0.3)
                wallJumpDebounce = false
            end
        end
    end
end)

UserInputService.JumpRequest:Connect(function()
    if Settings.InfinityJump then
        local ch = LocalPlayer.Character
        if ch then
            local hum = ch:FindFirstChildOfClass("Humanoid")
            if hum then hum:ChangeState(Enum.HumanoidStateType.Jumping) end
        end
    end
    if Settings.BunnyHop then
        local ch = LocalPlayer.Character
        if ch then
            local hu = ch:FindFirstChildOfClass("Humanoid")
            if hu then hu.Jump = true end
        end
    end
end)

LocalPlayer.CharacterAdded:Connect(function(ch)
    task.wait(0.5)
    local hu = ch:FindFirstChildOfClass("Humanoid")
    if hu then hu.AutoRotate = true end
    if Settings.ChamsEnabled then task.wait(0.5); ApplyChams(LocalPlayer) end
end)

-- Health tracking
local HealthTracker = {}
local function TrackHealth(player)
    if player == LocalPlayer then return end
    task.spawn(function()
        while player.Parent do
            local ch = player.Character
            if ch then
                local hu = ch:FindFirstChildOfClass("Humanoid")
                if hu then
                    local prev = HealthTracker[player] or hu.Health
                    if hu.Health < prev - 0.5 and Settings.DamageText then
                        local dmg = math.floor(prev - hu.Health)
                        local head = ch:FindFirstChild("Head")
                        if head and dmg > 0 then
                            ShowDamageText(head.Position, dmg)
                        end
                    end
                    HealthTracker[player] = hu.Health
                end
            end
            task.wait(0.1)
        end
    end)
end

for _, p in ipairs(Players:GetPlayers()) do TrackHealth(p) end
Players.PlayerAdded:Connect(TrackHealth)

-- ========== TOOL ESP UPDATE ==========
local function UpdateToolESP()
    if not Settings.ToolESP then
        for _, data in pairs(ToolESPData) do
            if data.hl then pcall(function() data.hl:Destroy() end) end
            if data.bb then pcall(function() data.bb:Destroy() end) end
        end
        ToolESPData = {}
        return
    end
    local foundTools = {}
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Tool") and not obj:FindFirstAncestorOfClass("Backpack") then
            local par = obj.Parent
            if par and not Players:GetPlayerFromCharacter(par) then
                foundTools[obj] = true
                if not ToolESPData[obj] then
                    ToolESPData[obj] = {}
                    local handle = obj:FindFirstChild("Handle") or obj:FindFirstChildWhichIsA("BasePart")
                    if handle then
                        local hl = Instance.new("Highlight")
                        hl.FillColor = C3(Settings.ToolESPColor)
                        hl.OutlineColor = C3(Settings.ToolESPColor)
                        hl.FillTransparency = 0.5
                        hl.OutlineTransparency = 0
                        hl.Adornee = obj
                        pcall(function() hl.Parent = obj end)
                        ToolESPData[obj].hl = hl
                        if Settings.ToolESPName then
                            local bb = Instance.new("BillboardGui")
                            bb.Size = UDim2.new(0, 100, 0, 30)
                            bb.StudsOffset = Vector3.new(0, 2, 0)
                            bb.AlwaysOnTop = true
                            bb.Adornee = handle
                            local lbl = Instance.new("TextLabel")
                            lbl.Size = UDim2.new(1, 0, 1, 0)
                            lbl.BackgroundTransparency = 1
                            lbl.Text = obj.Name
                            lbl.TextColor3 = C3(Settings.ToolESPNameColor)
                            lbl.TextSize = 12
                            lbl.Font = Enum.Font.GothamBold
                            lbl.TextStrokeTransparency = 0
                            lbl.Parent = bb
                            pcall(function() bb.Parent = handle end)
                            ToolESPData[obj].bb = bb
                        end
                    end
                else
                    if ToolESPData[obj].hl then
                        ToolESPData[obj].hl.FillColor = C3(Settings.ToolESPColor)
                        ToolESPData[obj].hl.OutlineColor = C3(Settings.ToolESPColor)
                    end
                end
            end
        end
    end
    for tool, data in pairs(ToolESPData) do
        if not foundTools[tool] or not tool.Parent then
            if data.hl then pcall(function() data.hl:Destroy() end) end
            if data.bb then pcall(function() data.bb:Destroy() end) end
            ToolESPData[tool] = nil
        end
    end
end

    else
        local ok, ping = pcall(function() return math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue()) end)
        PingLabel.Text = ok and ("Ping: " .. ping .. "ms") or "Ping: ?"
        PingLabel.Visible = Settings.ShowMS or Settings.ShowFPS
    end

    -- Movement
    local ch2 = LocalPlayer.Character
    if ch2 then
        local hu = ch2:FindFirstChildOfClass("Humanoid")
        local hrp = ch2:FindFirstChild("HumanoidRootPart")
        if hu and hrp then
            if Settings.BunnyHop and hu:GetState() == Enum.HumanoidStateType.Landed then
                hu.Jump = true
            end
            if Settings.CircleStrafe and hu.MoveDirection.Magnitude > 0 then
                hrp.CFrame = CFrame.new(hrp.Position) * CFrame.Angles(0, tt * 3, 0)
            end
            if Settings.StrafeInAir and hu:GetState() == Enum.HumanoidStateType.Freefall then
                local moveDir = hu.MoveDirection
                if moveDir.Magnitude > 0 then
                    hrp.Velocity = Vector3.new(moveDir.X * 30, hrp.Velocity.Y, moveDir.Z * 30)
                end
            end
            if Settings.QuickStop and hu.MoveDirection.Magnitude < 0.1 then
                hrp.Velocity = Vector3.new(0, hrp.Velocity.Y, 0)
            end
            if Settings.JumpBug and hu:GetState() == Enum.HumanoidStateType.Freefall and hrp.Velocity.Y < -30 then
                hu:ChangeState(Enum.HumanoidStateType.Landed)
            end
        end
    end

    -- Compass
    if Settings.Compass and ch2 then
        local dir = Camera.CFrame.LookVector
        local ang = math.deg(math.atan2(-dir.X, -dir.Z))
        local dirName = "N"
        if ang > -22.5 and ang <= 22.5 then dirName = "N"
        elseif ang > 22.5 and ang <= 67.5 then dirName = "NE"
        elseif ang > 67.5 and ang <= 112.5 then dirName = "E"
        elseif ang > 112.5 and ang <= 157.5 then dirName = "SE"
        elseif ang > 157.5 or ang <= -157.5 then dirName = "S"
        elseif ang > -157.5 and ang <= -112.5 then dirName = "SW"
        elseif ang > -112.5 and ang <= -67.5 then dirName = "W"
        elseif ang > -67.5 and ang <= -22.5 then dirName = "NW"
        end
        CompassLabel.Text = dirName .. " (" .. math.floor(ang) .. "°)"
    end

    -- Inventory Full Warning
    if Settings.InventoryFullWarning then
        local bp = LocalPlayer:FindFirstChild("Backpack")
        if bp then
            local count = 0
            for _, t in ipairs(bp:GetChildren()) do
                if t:IsA("Tool") then count = count + 1 end
            end
            InvFullFrame.Visible = count >= 6
            if count >= 6 then InvFullLabel.Text = "⚠ Inventory Full (" .. count .. ")" end
        end
    else
        InvFullFrame.Visible = false
    end

    -- TP Cooldown
    if Settings.TeleportCooldownInd then
        TPCooldownFrame.Visible = true
        TPCooldownLabel.Text = "TP: Ready"
        TPCooldownLabel.TextColor3 = Color3.fromRGB(80, 255, 120)
    end

    -- Nearby Enemy
    if Settings.NearbyEnemy and ch2 then
        local myHrp = ch2:FindFirstChild("HumanoidRootPart")
        if myHrp then
            local nearby = false
            local closestDist = math.huge
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                    local d = (p.Character.HumanoidRootPart.Position - myHrp.Position).Magnitude
                    if d < 30 then
                        nearby = true
                        if d < closestDist then closestDist = d end
                    end
                end
            end
            NearbyAlert.Visible = nearby
            if nearby then
                NearbyLabel.Text = "⚠ ENEMY " .. math.floor(closestDist) .. "m"
                NearbyStroke.Color = C3(Settings.NearbyEnemyColor)
                NearbyLabel.TextColor3 = C3(Settings.NearbyEnemyColor)
            end
        end
    else
        NearbyAlert.Visible = false
    end

    -- Tool ESP
    toolESPTimer = toolESPTimer + dt
    if toolESPTimer > 0.5 then
        toolESPTimer = 0
        UpdateToolESP()
    end

    -- Weather
    if Settings.WeatherEnabled and #WeatherParticles > 0 then
        for _, p in ipairs(WeatherParticles) do
            local pos = p.frame.Position
            local nx = pos.X.Scale + p.vx * dt
            local ny = pos.Y.Scale + p.vy * dt
            if ny > 1.1 then ny = -0.1; nx = math.random() end
            if nx > 1.1 then nx = 0 elseif nx < -0.1 then nx = 1 end
            p.frame.Position = UDim2.new(nx, 0, ny, 0)
        end
    end

    local speed = Settings.ParticleSpeed / 5
    if ParticleOverlay.Visible then
        for _, p in ipairs(BgParticles) do
            local pos = p.frame.Position
            local nx = pos.X.Scale + p.vx * dt * speed
            local ny = pos.Y.Scale + p.vy * dt * speed
            if nx < 0 then nx = 1 elseif nx > 1 then nx = 0 end
            if ny < 0 then ny = 1 elseif ny > 1 then ny = 0 end
            p.frame.Position = UDim2.new(nx, 0, ny, 0)
            if Settings.ParticleColor == "Rainbow" then
                p.hue = (p.hue + dt * 0.1) % 1
                local col = Color3.fromHSV(p.hue, 0.7, 1)
                p.frame.BackgroundColor3 = col
                p.stroke.Color = col
            end
        end
    end
    if MainFrame.Visible then
        for _, p in ipairs(InnerParticles) do
            local pos = p.frame.Position
            local nx = pos.X.Scale + p.vx * dt * speed
            local ny = pos.Y.Scale + p.vy * dt * speed
            if nx < 0 then nx = 1 elseif nx > 1 then nx = 0 end
            if ny < 0 then ny = 1 elseif ny > 1 then ny = 0 end
            p.frame.Position = UDim2.new(nx, 0, ny, 0)
        end
    end

    -- Title effects
    titleBeat = (titleBeat + dt * 145) % 360
    HeartLabel.Rotation = titleBeat
    HeartLabel.TextColor3 = Color3.fromHSV((titleBeat * 0.003 + 0.85) % 1, 0.8, 1)
    glowStroke.Transparency = 0.75 + math.sin(tt * 2) * 0.15
    glowStroke.Color = Color3.fromHSV((tt * 0.05 + 0.75) % 1, 0.7, 1)
    ToggleStroke.Thickness = 2.5 + math.sin(tt * 3) * 0.8

    -- FOV Circle
    if FOVCircle then
        FOVCircle.Position = center
        FOVCircle.Radius = Settings.FOV
        FOVCircle.Color = C3(Settings.FOVColor)
        FOVCircle.Visible = Settings.ShowFOV
    end

    -- ========== CROSSHAIR RENDERING ==========
    if Settings.CrosshairRotation then
        crosshairRot = (crosshairRot + dt * Settings.CrosshairRotSpeed) % 360
    end
    local rot = math.rad(crosshairRot)
    local sizeMult = Settings.CrosshairSize / 100
    local thickMult = Settings.CrosshairThickness / 100
    local opacity = Settings.CrosshairOpacity / 100
    local gap = Settings.CrosshairGap
    local cType = Settings.CrosshairType
    local cColor = C3(Settings.CrosshairColor)
    local oColor = C3(Settings.CrosshairOutlineColor)

    if cType == "Dot" then
        -- Single dot
        for i, d in ipairs(crosshairDots) do
            d.Position = center
            d.Radius = Settings.CrosshairDotSize * thickMult
            d.Color = cColor
            d.Transparency = opacity
            d.Visible = Settings.ShowCrosshair and opacity > 0.01
        end
        for _, l in ipairs(crosshairLines) do l.Visible = false end
        for _, l in ipairs(crosshairOutlineLines) do l.Visible = false end
    elseif cType == "Heart" or cType == "Square" then
        -- Dot-based shapes
        for i, d in ipairs(crosshairDots) do
            if crosshairPoints[i] then
                local pt = crosshairPoints[i]
                local scaled = Vector2.new(pt.X * sizeMult, pt.Y * sizeMult)
                if gap > 0 then
                    local dir = scaled.Magnitude > 0 and scaled.Unit or Vector2.new(0, 0)
                    scaled = scaled + dir * gap
                end
                local rv = RotateV2(scaled, rot)
                d.Position = center + rv
                d.Radius = 1.6 * thickMult
                d.Color = cColor
                d.Transparency = opacity
                d.Visible = Settings.ShowCrosshair and opacity > 0.01
            else
                d.Visible = false
            end
        end
        for _, l in ipairs(crosshairLines) do l.Visible = false end
        for _, l in ipairs(crosshairOutlineLines) do l.Visible = false end
    elseif cType == "Hexagon" or cType == "Star" then
        -- Line-based shapes
        for _, d in ipairs(crosshairDots) do d.Visible = false end
        local pts = crosshairPoints
        for i, l in ipairs(crosshairLines) do
            if pts[i] then
                local ni = (i % #pts) + 1
                local p1 = pts[i]
                local p2 = pts[ni]
                local s1 = Vector2.new(p1.X * sizeMult, p1.Y * sizeMult)
                local s2 = Vector2.new(p2.X * sizeMult, p2.Y * sizeMult)
                if gap > 0 then
                    local d1 = s1.Magnitude > 0 and s1.Unit or Vector2.new(0, 0)
                    local d2 = s2.Magnitude > 0 and s2.Unit or Vector2.new(0, 0)
                    s1 = s1 + d1 * gap
                    s2 = s2 + d2 * gap
                end
                local r1 = RotateV2(s1, rot)
                local r2 = RotateV2(s2, rot)
                l.From = center + r1
                l.To = center + r2
                l.Color = cColor
                l.Thickness = 2 * thickMult
                l.Transparency = opacity
                l.Visible = Settings.ShowCrosshair and opacity > 0.01
            else
                l.Visible = false
            end
        end
        -- Outline
        for i, ol in ipairs(crosshairOutlineLines) do
            if Settings.CrosshairOutline and pts[i] and crosshairLines[i] then
                local ni = (i % #pts) + 1
                local p1 = pts[i]
                local p2 = pts[ni]
                local s1 = Vector2.new(p1.X * sizeMult, p1.Y * sizeMult)
                local s2 = Vector2.new(p2.X * sizeMult, p2.Y * sizeMult)
                if gap > 0 then
                    local d1 = s1.Magnitude > 0 and s1.Unit or Vector2.new(0, 0)
                    local d2 = s2.Magnitude > 0 and s2.Unit or Vector2.new(0, 0)
                    s1 = s1 + d1 * gap
                    s2 = s2 + d2 * gap
                end
                local r1 = RotateV2(s1, rot)
                local r2 = RotateV2(s2, rot)
                ol.From = center + r1
                ol.To = center + r2
                ol.Color = oColor
                ol.Thickness = (2 * thickMult) + 2
                ol.Transparency = opacity * 0.7
                ol.Visible = Settings.ShowCrosshair and opacity > 0.01
            else
                if ol then ol.Visible = false end
            end
        end
    end

    -- ========== AIMBOT ==========
    local shouldAim = false
    if Settings.AimbotEnabled then
        if Settings.AimbotMode == "Always" then
            shouldAim = true
        elseif Settings.AimbotMode == "Hold" then
            shouldAim = AimbotActive
        elseif Settings.AimbotMode == "Toggle" then
            shouldAim = ToggleActive
        end
    end
    if shouldAim then
        local target, pos = GetClosestPlayer()
        if target and pos then
            local tcf = CFrame.new(Camera.CFrame.Position, pos)
            if Settings.Smoothness <= 0 then
                Camera.CFrame = tcf
            else
                local a = math.clamp(dt * (22 - Settings.Smoothness) * 3, 0, 1)
                Camera.CFrame = Camera.CFrame:Lerp(tcf, a)
            end
        end
    end

    -- ========== RADAR ==========
    if Settings.RadarEnabled and RadarFrame.Visible then
        local lp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        for p, dot in pairs(RadarDots) do
            if not p.Parent or not p.Character then
                dot.Visible = false
            else
                local root = p.Character:FindFirstChild("HumanoidRootPart")
                if root and lp then
                    local rel = root.Position - lp.Position
                    local camCF = Camera.CFrame
                    local look = Vector3.new(camCF.LookVector.X, 0, camCF.LookVector.Z).Unit
                    local ang = math.atan2(look.X, look.Z)
                    local cs, sn = math.cos(-ang), math.sin(-ang)
                    local rx = rel.X * cs - rel.Z * sn
                    local rz = rel.X * sn + rel.Z * cs
                    local zoom = Settings.RadarZoom
                    local px = math.clamp(0.5 + (rx / zoom / 2), 0, 1)
                    local py = math.clamp(0.5 + (rz / zoom / 2), 0, 1)
                    dot.Position = UDim2.new(px, -3, py, -3)
                    dot.BackgroundColor3 = C3(Settings.ESPColor)
                    dot.Visible = true
                else
                    dot.Visible = false
                end
            end
        end
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer then GetRadarDot(p) end
        end
    else
        for _, dot in pairs(RadarDots) do dot.Visible = false end
    end

    -- ========== ESP ==========
    for player, esp in pairs(ESPData) do
        if not player or not player.Parent then HideESP(esp); continue end
        if not Settings.ESPEnabled then HideESP(esp); continue end
        local ch3 = player.Character
        if not ch3 then
            if Settings.ESPDormant and dormantPositions[player] and esp.DormantMarker then
                local dp = dormantPositions[player]
                local sp, onS = Camera:WorldToViewportPoint(dp)
                if onS then
                    esp.DormantMarker.From = Vector2.new(sp.X, sp.Y - 10)
                    esp.DormantMarker.To = Vector2.new(sp.X, sp.Y + 10)
                    esp.DormantMarker.Color = Color3.fromRGB(150, 150, 150)
                    esp.DormantMarker.Visible = true
                end
            else
                HideESP(esp)
            end
            continue
        end
        local hu = ch3:FindFirstChildOfClass("Humanoid")
        local root = ch3:FindFirstChild("HumanoidRootPart")
        local head = ch3:FindFirstChild("Head")
        if not (hu and hu.Health > 0 and root and head) then HideESP(esp); continue end
        dormantPositions[player] = root.Position
        local ht = head.Position + Vector3.new(0, head.Size.Y / 2, 0)
        local rb = root.Position - Vector3.new(0, root.Size.Y / 2 + 2.3, 0)
        local hs, hv = Camera:WorldToViewportPoint(ht)
        local rs, rv = Camera:WorldToViewportPoint(rb)
        if not (hv and rv) then HideESP(esp); continue end
        local top, bot = hs.Y, rs.Y
        local h = bot - top
        local cx = (hs.X + rs.X) / 2
        local w = math.max(h * 0.5, 20)

        for _, l in ipairs(esp.BoxLines) do l.Visible = false end
        for _, l in ipairs(esp.CornerLines) do l.Visible = false end
        for _, l in ipairs(esp.Box3DLines) do l.Visible = false end

        if Settings.ESPBoxes then
            if Settings.ESPBoxStyle == "2D" then
                local tl = Vector2.new(cx - w / 2, top)
                local tr2 = Vector2.new(cx + w / 2, top)
                local bl = Vector2.new(cx - w / 2, bot)
                local br = Vector2.new(cx + w / 2, bot)
                local pts = {tl, tr2, br, bl}
                for i = 1, 4 do
                    local ni = (i % 4) + 1
                    if esp.BoxLines[i] then
                        esp.BoxLines[i].From = pts[i]
                        esp.BoxLines[i].To = pts[ni]
                        esp.BoxLines[i].Color = C3(Settings.ESPColor)
                        esp.BoxLines[i].Visible = true
                    end
                end
            elseif Settings.ESPBoxStyle == "Corner" then
                local cornerLen = math.min(w, h) * 0.25
                local pts = {
                    {Vector2.new(cx - w / 2, top), Vector2.new(cx - w / 2 + cornerLen, top)},
                    {Vector2.new(cx - w / 2, top), Vector2.new(cx - w / 2, top + cornerLen)},
                    {Vector2.new(cx + w / 2, top), Vector2.new(cx + w / 2 - cornerLen, top)},
                    {Vector2.new(cx + w / 2, top), Vector2.new(cx + w / 2, top + cornerLen)},
                    {Vector2.new(cx - w / 2, bot), Vector2.new(cx - w / 2 + cornerLen, bot)},
                    {Vector2.new(cx - w / 2, bot), Vector2.new(cx - w / 2, bot - cornerLen)},
                    {Vector2.new(cx + w / 2, bot), Vector2.new(cx + w / 2 - cornerLen, bot)},
                    {Vector2.new(cx + w / 2, bot), Vector2.new(cx + w / 2, bot - cornerLen)},
                }
                for i = 1, 8 do
                    if esp.CornerLines[i] then
                        esp.CornerLines[i].From = pts[i][1]
                        esp.CornerLines[i].To = pts[i][2]
                        esp.CornerLines[i].Color = C3(Settings.ESPColor)
                        esp.CornerLines[i].Thickness = 2
                        esp.CornerLines[i].Visible = true
                    end
                end
            elseif Settings.ESPBoxStyle == "3D" then
                local ext2 = Vector3.new(2, 3, 1.5)
                local corners = {
                    root.Position + Vector3.new(-ext2.X, ext2.Y, -ext2.Z),
                    root.Position + Vector3.new(ext2.X, ext2.Y, -ext2.Z),
                    root.Position + Vector3.new(ext2.X, ext2.Y, ext2.Z),
                    root.Position + Vector3.new(-ext2.X, ext2.Y, ext2.Z),
                    root.Position + Vector3.new(-ext2.X, -ext2.Y, -ext2.Z),
                    root.Position + Vector3.new(ext2.X, -ext2.Y, -ext2.Z),
                    root.Position + Vector3.new(ext2.X, -ext2.Y, ext2.Z),
                    root.Position + Vector3.new(-ext2.X, -ext2.Y, ext2.Z),
                }
                local screenP = {}
                local allOn = true
                for i, cor in ipairs(corners) do
                    local sp, on = Camera:WorldToViewportPoint(cor)
                    screenP[i] = Vector2.new(sp.X, sp.Y)
                    if not on then allOn = false end
                end
                if allOn then
                    local edges = {{1, 2}, {2, 3}, {3, 4}, {4, 1}, {5, 6}, {6, 7}, {7, 8}, {8, 5}, {1, 5}, {2, 6}, {3, 7}, {4, 8}}
                    for i, e in ipairs(edges) do
                        if esp.Box3DLines[i] then
                            esp.Box3DLines[i].From = screenP[e[1]]
                            esp.Box3DLines[i].To = screenP[e[2]]
                            esp.Box3DLines[i].Color = C3(Settings.ESPColor)
                            esp.Box3DLines[i].Visible = true
                        end
                    end
                end
            end
        end

        -- Skeleton
        for _, l in ipairs(esp.SkeletonLines) do l.Visible = false end
        if Settings.ESPSkeleton then
            for i, bone in ipairs(SkeletonBones) do
                local p1 = ch3:FindFirstChild(bone[1])
                local p2 = ch3:FindFirstChild(bone[2])
                if p1 and p2 and esp.SkeletonLines[i] then
                    local s1, o1 = Camera:WorldToViewportPoint(p1.Position)
                    local s2, o2 = Camera:WorldToViewportPoint(p2.Position)
                    if o1 and o2 then
                        esp.SkeletonLines[i].From = Vector2.new(s1.X, s1.Y)
                        esp.SkeletonLines[i].To = Vector2.new(s2.X, s2.Y)
                        esp.SkeletonLines[i].Color = C3(Settings.SkeletonColor)
                        esp.SkeletonLines[i].Visible = true
                    end
                end
            end
        end

        -- Name
        if Settings.ESPNames and esp.NameText then
            local nameStr = player.Name
            if Settings.ShowPlayerFPS then nameStr = nameStr .. " [FPS:" .. currentFPS .. "]" end
            if Settings.ShowPlayerMS then nameStr = nameStr .. " [MS:" .. math.floor(dt * 1000) .. "]" end
            esp.NameText.Text = nameStr
            esp.NameText.Position = Vector2.new(cx, top - 18)
            esp.NameText.Color = C3(Settings.ESPColor)
            esp.NameText.Visible = true
        elseif esp.NameText then
            esp.NameText.Visible = false
        end

        -- Distance
        if Settings.ESPDistance and esp.DistText and LocalPlayer.Character then
            local lp = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
            if lp then
                local d = math.floor((lp.Position - root.Position).Magnitude)
                esp.DistText.Text = "[" .. d .. "m]"
                esp.DistText.Position = Vector2.new(cx, bot + 4)
                esp.DistText.Visible = true
            end
        elseif esp.DistText then
            esp.DistText.Visible = false
        end

        -- Weapon
        if Settings.ESPWeapon and esp.WeaponText then
            local tool = ch3:FindFirstChildOfClass("Tool")
            if tool then
                esp.WeaponText.Text = "[" .. tool.Name .. "]"
                esp.WeaponText.Position = Vector2.new(cx, bot + 18)
                esp.WeaponText.Color = Color3.fromRGB(255, 200, 100)
                esp.WeaponText.Visible = true
            else
                esp.WeaponText.Visible = false
            end
        elseif esp.WeaponText then
            esp.WeaponText.Visible = false
        end

        -- Health Bar
        if Settings.ESPHealth and esp.HpBG and esp.HpFill then
            local hp = math.clamp(hu.Health / hu.MaxHealth, 0, 1)
            local bx = cx - w / 2 - 7
            esp.HpBG.From = Vector2.new(bx, top)
            esp.HpBG.To = Vector2.new(bx, bot)
            esp.HpBG.Visible = true
            esp.HpFill.Color = Color3.fromRGB(math.floor((1 - hp) * 220), math.floor(hp * 220), 30)
            esp.HpFill.From = Vector2.new(bx, bot)
            esp.HpFill.To = Vector2.new(bx, bot - h * hp)
            esp.HpFill.Visible = true
        else
            if esp.HpBG then esp.HpBG.Visible = false end
            if esp.HpFill then esp.HpFill.Visible = false end
        end

        -- Tracers
        if Settings.ESPTracers and esp.Tracer then
            esp.Tracer.Color = C3(Settings.ESPColor)
            esp.Tracer.From = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
            esp.Tracer.To = Vector2.new(cx, bot)
            esp.Tracer.Visible = true
        elseif esp.Tracer then
            esp.Tracer.Visible = false
        end

        -- Snap Lines
        if Settings.ESPSnapLines and esp.SnapLine then
            esp.SnapLine.Color = C3(Settings.ESPColor)
            esp.SnapLine.From = center
            esp.SnapLine.To = Vector2.new(cx, (top + bot) / 2)
            esp.SnapLine.Transparency = 0.5
            esp.SnapLine.Visible = true
        elseif esp.SnapLine then
            esp.SnapLine.Visible = false
        end

        if esp.DormantMarker then esp.DormantMarker.Visible = false end
    end
end)

TitleBar.InputBegan:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        mDr = true
        mDs = i.Position
        mSp = MainFrame.Position
    end
end)

    if mDr and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
        local d = i.Position - mDs
        MainFrame.Position = UDim2.new(mSp.X.Scale, mSp.X.Offset + d.X, mSp.Y.Scale, mSp.Y.Offset + d.Y)
    end
end)
UserInputService.InputEnded:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then mDr = false end
end)

ToggleButton.InputBegan:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        tDr = true
        tMv = false
        tDs = i.Position
        tSp = ToggleButton.Position
    end
end)
UserInputService.InputChanged:Connect(function(i)
    if tDr and not Settings.ToggleButtonLocked and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
        local d = i.Position - tDs
        if d.Magnitude > 7 then
            tMv = true
            ToggleButton.Position = UDim2.new(tSp.X.Scale, tSp.X.Offset + d.X, tSp.Y.Scale, tSp.Y.Offset + d.Y)
        end
    end
end)
UserInputService.InputEnded:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then tDr = false end
end)

ToggleButton.MouseButton1Click:Connect(function()
    if tMv then tMv = false; return end
    guiOpen = not guiOpen
    MainFrame.Visible = guiOpen
    ParticleOverlay.Visible = guiOpen
    if guiOpen then
        MainFrame.BackgroundTransparency = 1
        TweenService:Create(MainFrame, TweenInfo.new(0.3, Enum.EasingStyle.Back), {BackgroundTransparency = 0}):Play()
    end
end)

CloseButton.MouseButton1Click:Connect(function()
    guiOpen = false
